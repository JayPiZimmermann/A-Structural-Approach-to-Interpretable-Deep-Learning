\begin{definition}[Feedforward \ac{relu} network]\label{def:ffn_relu}
Let $L \in \NN$ and let $d^{(0)}, d^{(1)}, \ldots, d^{(L)} \in \NN$ denote the \emph{layer widths}, with input dimension $d^{(0)}$ and output dimension $d^{(L)}$. A \emph{feedforward \ac{relu} network} of depth $L$ is given by weight matrices $\Wl{l} \in \RR^{d^{(l)} \times d^{(l-1)}}$ and bias vectors $b^{(l)} \in \RR^{d^{(l)}}$ for $l = 1, \ldots, L$. On an input $x \in \RR^{d^{(0)}}$ it computes the recursion
\begin{align}
	\actl{0} &\coloneqq x, \notag\\
	\preactl{l} &\coloneqq \Wl{l} \actl{l-1} + b^{(l)}, && l = 1, \ldots, L, \label{eq:forward_recursion}\\
	\actl{l} &\coloneqq \relu\bigl(\preactl{l}\bigr), && l = 1, \ldots, L-1, \notag
\end{align}
where $\relu(z) \coloneqq \max\{z, 0\}$ acts componentwise, and the output layer is affine, $\actl{L} \coloneqq \preactl{L}$. We call $\preactl{l}$ the \emph{pre-activation} and $\actl{l}$ the \emph{(post-)activation} of layer $l$, and denote the \emph{network function} by
\begin{equation}
	f \colon \RR^{d^{(0)}} \longrightarrow \RR^{d^{(L)}},
	\qquad
	f(x) \coloneqq \actl{L}.
	\label{eq:network_function}
\end{equation}
Layers $1, \ldots, L-1$ are the \emph{hidden layers}. Throughout we fix the boundary convention $\relu'(0) \coloneqq 0$, so that the gate indicator $\1\{\preactl{l}_i > 0\}$ returns $0$ at $\preactl{l}_i = 0$.
\end{definition}

\begin{definition}[Positive and negative parts]\label{def:posneg_parts}
For a scalar $u \in \RR$ we write $u^{+} \coloneqq \max\{u, 0\}$ and $u^{-} \coloneqq \max\{-u, 0\}$, so that $u = u^{+} - u^{-}$ and $\abs{u} = u^{+} + u^{-}$, with $u^+ u^- = 0$. For a matrix or vector $W$ and a sign label $\sigma \in \{+,-\}$, the \emph{entrywise part} $\posneg{W}{\sigma}$ is obtained by applying $u \mapsto u^{\sigma}$ to every entry. For the layer weights and biases of Definition~\ref{def:ffn_relu} we abbreviate
\begin{equation}
	\Wsig{l}{\sigma} \coloneqq \posneg{\Wl{l}}{\sigma},
	\qquad
	b^{(l,\sigma)} \coloneqq \posneg{b^{(l)}}{\sigma},
	\qquad \sigma \in \{+,-\},
	\label{eq:weight_split}
\end{equation}
so that $\Wl{l} = \Wsig{l}{+} - \Wsig{l}{-}$ and $\abs{\Wl{l}} = \Wsig{l}{+} + \Wsig{l}{-}$ entrywise, and both $\Wsig{l}{\sigma}$ are entrywise non-negative.
\end{definition}

\begin{definition}[\ac{cpwl} function]\label{def:cpwl}
A function $f \colon \RR^{d} \to \RR^{m}$ is \emph{\acl{cpwl}} (\acs{cpwl}) if it is continuous and there exist finitely many closed polyhedra $Q_1, \ldots, Q_N \subseteq \RR^d$ with $\bigcup_{r=1}^{N} Q_r = \RR^d$ such that the restriction of $f$ to each $Q_r$ is an affine map. The full-dimensional polyhedra on which $f$ is affine, taken maximal with respect to inclusion, are called the \emph{linear regions} of $f$.
\end{definition}

\begin{remark}[Monotonicity and convexity combine]\label{rem:monotone_convex_combined}
If, in addition, $\Wl{1} \ge 0$ in Definition~\ref{def:icnn}, the network is simultaneously monotone and input-convex, hence non-decreasing \emph{and} convex by Propositions~\ref{prop:monotone_represents} and \ref{prop:icnn_convex}. \end{remark}

\begin{definition}[Maxout unit and Maxout network, \parencite{Goodfellow2013Maxout}]\label{def:maxout}
A \emph{rank-$k$ Maxout unit} with weights $w_1, \ldots, w_k \in \RR^{d}$ and biases $\beta_1, \ldots, \beta_k \in \RR$ computes
\begin{equation}
	x \longmapsto \maxout(x) \coloneqq \max_{j \in [k]} \bigl( w_j^{\top} x + \beta_j \bigr).
	\label{eq:maxout_unit}
\end{equation}
A \emph{rank-$k$ Maxout network} is a feedforward network whose neurons are rank-$k$ Maxout units; equivalently, layer $l$ carries $k$ weight matrices $W^{(l)}_1, \ldots, W^{(l)}_k$ and biases $b^{(l)}_1, \ldots, b^{(l)}_k$, and
\begin{equation}
	\actl{l}_i \coloneqq \max_{j \in [k]} \bigl( W^{(l)}_j \actl{l-1} + b^{(l)}_j \bigr)_i .
	\label{eq:maxout_layer}
\end{equation}
\end{definition}

\begin{definition}[\acl{mdp}]\label{def:mdp}
A (finite) \emph{\acl{mdp}} (\acs{mdp}) \parencite{Puterman1994mdp,SuttonBarto2018rl} is a tuple $\mathcal{M} = (S, A, \P, R, \gamma)$ consisting of a finite state set $S$, a finite action set $A(s)$ for each $s \in S$, a \emph{transition kernel} $\P(s' \mid s, a) \ge 0$ with $\sum_{s'} \P(s' \mid s, a) = 1$, a bounded \emph{reward function} $R(s, a) \in \RR$, and a \emph{discount factor} $\gamma \in [0, 1)$. A \emph{(stationary) policy} is a map $\pi$ assigning to each state $s$ a probability distribution $\pi(\cdot \mid s)$ over $A(s)$; $\pi$ is \emph{pure} (deterministic) if each $\pi(\cdot \mid s)$ is a point mass. A policy induces, for every start state, a Markov chain on $S$ and hence a distribution over trajectories $(s_0, a_0, s_1, a_1, \ldots)$; the \emph{value function} of $\pi$ is the expected discounted return
\begin{equation}
	V^{\pi}(s) \coloneqq \E^{\pi}\Bigl[ \textstyle\sum_{t=0}^{\infty} \gamma^{t}\, R(s_t, a_t) \,\Big|\, s_0 = s \Bigr],
	\label{eq:mdp_value}
\end{equation}
and the \emph{optimal value function} is $V^{\star}(s) \coloneqq \max_{\pi} V^{\pi}(s)$.
\end{definition}

\begin{proposition}[Bellman equations, \LeanResult{prop:bellman}]\label{prop:bellman}
Let $\mathcal{M}$ be a finite \ac{mdp} with $\gamma \in [0,1)$. Then:
\begin{enumerate}
	\item $V^{\pi}$ is the unique solution of the linear system
	\begin{equation}
		V(s) = \sum_{a \in A(s)} \pi(a \mid s) \Bigl[ R(s,a) + \gamma \sum_{s'} \P(s' \mid s, a)\, V(s') \Bigr],
		\qquad s \in S.
		\label{eq:bellman_evaluation}
	\end{equation}
	\item $V^{\star}$ is the unique solution of the \emph{Bellman optimality equation}
	\begin{equation}
		V(s) = \max_{a \in A(s)} \Bigl\{ R(s, a) + \gamma \sum_{s'} \P(s' \mid s, a)\, V(s') \Bigr\},
		\qquad s \in S,
		\label{eq:bellman_optimality}
	\end{equation}
	and any pure policy $\polstar$ that selects, in every state, an action attaining the maximum in \eqref{eq:bellman_optimality} is optimal: $V^{\polstar} = V^{\star}$.
\end{enumerate}\end{proposition}

\begin{remark}[State-dependent structural discounts]\label{rem:structural_discount}
The games of this thesis depart from Definition~\ref{def:mdp} in one respect: the discount is \emph{state-dependent}, $\gamma(s) \ge 0$, and may exceed $1$. Here discounts are not impatience parameters but \emph{structural} normalizers---for neuron $i$ in layer $l$ the absolute weight mass
\begin{equation}
	\gamma^{(l)}_i \coloneqq \sum_{j} \bigl( \Wsig{l}{+} + \Wsig{l}{-} \bigr)_{ij} = \sum_j \abs{\Wl{l}_{ij}},
	\label{eq:structural_discount}
\end{equation}
which restores the weight mass after row $i$ of $\abs{\Wl{l}}$ is normalized into transition probabilities. Contraction is then unavailable, but also unnecessary: all games are played on \emph{layered acyclic} graphs, so every trajectory is finite and all value functions are well-defined finite sums. Backward induction gives a unique value and the existence of optimal policies; policies need not be unique when actions tie. The two-player statement follows below.
\end{remark}

\begin{definition}[Zero-sum turn-based game]\label{def:turn_based_game}
A (finite) \emph{zero-sum turn-based game} is a tuple $\mathcal{G} = (S_{+}, S_{-}, S_{\mathrm{ter}}, A, \P, R, \gamma)$ where the state set $S = S_{+} \uplus S_{-} \uplus S_{\mathrm{ter}}$ is partitioned into states controlled by player $P^{+}$, states controlled by player $P^{-}$, and terminal states; the action sets $A(s)$, the transition kernel $\P$, and the reward $R$ are as in Definition~\ref{def:mdp} on non-terminal states; $\gamma \colon S \to \RR_{\ge 0}$ is a (possibly state-dependent) discount; and each terminal state $s \in S_{\mathrm{ter}}$ carries a fixed one-time payoff $R(s)$. A \emph{policy pair} $\pi = (\pi_{+}, \pi_{-})$ consists of a policy $\pi_{p}$ for player $P^{p}$ on $S_{p}$. Play generates a trajectory as in an \ac{mdp}, with the player in turn choosing the action; the expected total payoff $V^{\pi}(s)$, defined as in \eqref{eq:mdp_value} with the discount accumulated multiplicatively along the trajectory ($\gamma^t$ replaced by $\prod_{m=0}^{t-1}\gamma(s_m)$), is paid \emph{by $P^{-}$ to $P^{+}$}: player $P^{+}$ maximizes $V^{\pi}$ and player $P^{-}$ minimizes it.
\end{definition}

30 tex/chapters/02_foundations.tex:216 theoremenv ['thm:shapley_bellman'] LINKED
\begin{theoremenv}[Shapley--Bellman optimality on layered games, \LeanResult{thm:shapley_bellman}]\label{thm:shapley_bellman}
Let $\mathcal{G}$ be a finite zero-sum turn-based game whose transition graph is acyclic. Then $\mathcal{G}$ has a \emph{value}: there is a unique function $V^{\star} \colon S \to \RR$ satisfying the \emph{Shapley--Bellman optimality equations}
\begin{equation}
	V^{\star}(s) =
	\begin{cases}
		\displaystyle \max_{a \in A(s)} \Bigl\{ R(s,a) + \gamma(s) \sum_{s'} \P(s' \mid s, a)\, V^{\star}(s') \Bigr\}, & s \in S_{+}, \\[1.2em]
		\displaystyle \min_{a \in A(s)} \Bigl\{ R(s,a) + \gamma(s) \sum_{s'} \P(s' \mid s, a)\, V^{\star}(s') \Bigr\}, & s \in S_{-}, \\[1.2em]
		R(s), & s \in S_{\mathrm{ter}},
	\end{cases}
	\label{eq:shapley_bellman}
\end{equation}
and there exist pure stationary policies $\polstar = (\polstar_{+}, \polstar_{-})$, obtained by selecting maximizing respectively minimizing actions in \eqref{eq:shapley_bellman}, such that for all $s \in S$ and all policies $\pi_{+}, \pi_{-}$,
\begin{equation}
	V^{(\polstar_{+}, \pi_{-})}(s) \;\ge\; V^{\polstar}(s) = V^{\star}(s) \;\ge\; V^{(\pi_{+}, \polstar_{-})}(s).
	\label{eq:saddle_point}
\end{equation}
In particular, $\polstar$ is a Nash equilibrium of every subgame, i.e.\ a subgame perfect equilibrium.\end{theoremenv}

\begin{definition}[ReLU Net Game, following \parencite{Gaubert2025}]\label{def:relu_net_game}
The \emph{ReLU Net Game} on the network $f$ at input $x$ is the zero-sum turn-based game with the following components.
\begin{itemize}
	\item \textbf{States.} For every hidden layer $l \in \{1, \ldots, L-1\}$, every neuron $i \in [d^{(l)}]$, and every turn label $q \in \{+,-\}$ there is an \emph{activation state} $s^{(l)}_{i,q}$, controlled by player $P^{q}$, so that $S_q = \bigl\{ s^{(l)}_{i,q} : l \in [L-1],\, i \in [d^{(l)}] \bigr\}$. For every input coordinate $k \in [d^{(0)}]$ and every $q \in \{+,-\}$ there is a \emph{terminal state} $s^{(0)}_{k,q}$, and there is one absorbing \emph{cemetery state} $\perp$ with zero payoff. Finally, there is a single \emph{output state} $s^{(L)}$ at the affine output neuron, at which no decision is taken.
	\item \textbf{Actions.} At every activation state the player in turn chooses an action from $\{\mathrm{stop}, \mathrm{cont}\}$. The output state has the single action $\mathrm{cont}$.
	\item \textbf{Transitions.} Stopping moves deterministically to the cemetery, $\P(\perp \mid s^{(l)}_{i,q}, \mathrm{stop}) = 1$. Continuation at $s^{(l)}_{i,q}$ moves one layer backward, to neuron $j$ of layer $l-1$ with probability proportional to the absolute weight, keeping the turn on positive weights and passing it to the opponent on negative weights:
	\begin{equation}
		\P\bigl( s^{(l-1)}_{j,r} \,\big|\, s^{(l)}_{i,q}, \mathrm{cont} \bigr)
		\coloneqq \frac{\Wsig{l}{q \oplus r}_{ij}}{\gamma^{(l)}_i},
		\qquad
		\gamma^{(l)}_i = \sum_{j} \bigl( \Wsig{l}{+} + \Wsig{l}{-} \bigr)_{ij},
		\label{eq:rng_kernel}
	\end{equation}
	where for $l = 1$ the successor $s^{(0)}_{j,r}$ is the corresponding terminal state. The output state transitions by the same rule \eqref{eq:rng_kernel} with $l = L$ and turn label $q = +$.
	\item \textbf{Payoffs and discounts.} Continuation at $s^{(l)}_{i,q}$ pays the signed bias $\xi_{q}\, b^{(l)}_i$; stopping pays $0$. A terminal state pays the signed input coordinate once, $R\bigl(s^{(0)}_{k,q}\bigr) \coloneqq \xi_{q}\, x_k$, and the game ends. The output state pays $b^{(L)}$ upon its forced continuation. The discount at every activation state and at the output state is the structural discount $\gamma\bigl(s^{(l)}_{i,q}\bigr) \coloneqq \gamma^{(l)}_i$ of \eqref{eq:structural_discount}.
\end{itemize}
Player $P^{+}$ maximizes and $P^{-}$ minimizes the expected discounted total payoff, as in Definition~\ref{def:turn_based_game}.
\end{definition}

\begin{remark}[Softplus networks from entropy regularization]\label{rem:softplus_game}
\textcite{Gaubert2025} also treat softplus networks, by \emph{regularizing} the stop/continue decision. Replace the hard choice at an activation state by a mixed policy $\nu \in \Delta(\{\mathrm{stop}, \mathrm{cont}\})$ rewarded with a Shannon-entropy bonus at temperature $\theta > 0$. By the Legendre--Fenchel identity for the entropy \parencite{BoydVandenberghe2004},
\begin{equation}
	\max_{\nu \in \Delta_n} \Bigl\{ \textstyle\sum_{i=1}^{n} \nu_i r_i + \theta\, \mathcal{H}(\nu) \Bigr\}
	= \theta \log \textstyle\sum_{i=1}^{n} e^{r_i / \theta},
	\qquad
	\mathcal{H}(\nu) = -\textstyle\sum_i \nu_i \log \nu_i,
	\label{eq:legendre_fenchel}
\end{equation}
with the Gibbs distribution $\nu^{\star}_i \propto e^{r_i/\theta}$ as unique maximizer, the regularized value at a state with continuation value $\preactl{l}_i$ (computed as in \eqref{eq:rng_cont_value}) becomes $\theta \log\bigl(1 + e^{\preactl{l}_i / \theta}\bigr)$---the softplus activation at temperature $\theta$---and the equilibrium policy continues with the sigmoid probability $\bigl(1 + e^{-\preactl{l}_i/\theta}\bigr)^{-1}$. The hard game of Definition~\ref{def:relu_net_game} is recovered in the limit $\theta \to 0$. Chapter~\ref{ch:game_theory} re-derives this mechanism from the regularized decompositions of Section~\ref{sec:regularized_decompositions} and reuses it at the routing states of the \ac{rg}, with the two temperatures $\theta$ (activation gates) and $\tau$ (routing) kept distinct.
\end{remark}

\begin{definition}[Max-plus and min-plus semirings]\label{def:maxplus_semiring}
The \emph{max-plus semiring} is the set $\RR_{\max} \coloneqq \RR \cup \{-\infty\}$ equipped with \emph{tropical addition} and \emph{tropical multiplication},
\begin{equation}
	u \oplus v \coloneqq \max\{u, v\},
	\qquad
	u \odot v \coloneqq u + v,
	\label{eq:tropical_operations}
\end{equation}
with additive neutral element $-\infty$ and multiplicative neutral element $0$. Both operations are associative and commutative, and $\odot$ distributes over $\oplus$; additive inverses do not exist, so $\RR_{\max}$ is a semiring but not a ring. The \emph{min-plus semiring} $\RR_{\min} \coloneqq \RR \cup \{+\infty\}$ is defined dually with $u \oplus' v \coloneqq \min\{u,v\}$; the map $u \mapsto -u$ is an isomorphism between the two, so every statement below has a min-plus counterpart.
\end{definition}

45 tex/chapters/02_foundations.tex:352 definition ['def:tropical_polynomial'] MISSING
\begin{definition}[Tropical polynomials and rational functions]\label{def:tropical_polynomial}
A \emph{tropical monomial} in $d$ variables with exponent $\alpha \in \RR^{d}$ and coefficient $c \in \RR$ is the affine function $c \odot x_1^{\odot \alpha_1} \odot \cdots \odot x_d^{\odot \alpha_d} = c + \langle \alpha, x \rangle$. A \emph{tropical polynomial} with finite support $\mathcal{A} \subset \RR^{d}$ and coefficients $(c_{\alpha})_{\alpha \in \mathcal{A}}$, $c_\alpha \in \RR$, is the tropical sum of monomials,
\begin{equation}
	p(x) \;=\; \bigoplus_{\alpha \in \mathcal{A}} c_{\alpha} \odot x^{\odot \alpha}
	\;=\; \max_{\alpha \in \mathcal{A}} \bigl( c_{\alpha} + \langle \alpha, x \rangle \bigr).
	\label{eq:tropical_polynomial}
\end{equation}
A \emph{tropical rational function} is a tropical quotient of two tropical polynomials, $p \oslash q \coloneqq p - q$. In classical tropical geometry \parencite{MaclaganSturmfels2015tropical} the exponents are restricted to $\mathcal{A} \subset \NN^{d}$ or $\ZZ^{d}$; for neural networks, real exponents (network weights) are the natural setting, and all statements below hold verbatim in this generality.
\end{definition}

\begin{lemmaenv}[Properties of support functions, \LeanResult{lem:support_properties}]\label{lem:support_properties}
Let $P, Q \subset \RR^{d}$ be nonempty compact convex sets. Then:
\begin{enumerate}
	\item $h_{P}$ is convex and positively homogeneous, i.e.\ $h_P(\lambda x) = \lambda h_P(x)$ for $\lambda \ge 0$; if $P$ is a polytope with vertex set $\operatorname{vert}(P)$, then $h_{P}(x) = \max_{u \in \operatorname{vert}(P)} \langle u, x \rangle$ is a tropical polynomial with zero coefficients and Newton polytope $P$.
	\item For any nonempty finite set $A \subset \RR^d$ and $x \in \RR^d$, $\max_{u \in \operatorname{conv}(A)} \langle u, x \rangle = \max_{u \in A} \langle u, x \rangle$; in particular $h_{P + Q} = h_{P} + h_{Q}$ and $h_{\operatorname{conv}(P \cup Q)} = \max\{h_{P}, h_{Q}\} = h_P \oplus h_Q$.
	\item $P \subseteq Q$ if and only if $h_{P} \le h_{Q}$ pointwise; in particular $h_{P} = h_{Q}$ implies $P = Q$, so a compact convex set is uniquely determined by its support function.
\end{enumerate}\end{lemmaenv}

\begin{proposition}[Newton polytope calculus, cf.\ \parencite{Zhang2018}, \LeanResult{prop:newton_calculus}]\label{prop:newton_calculus}

\begin{equation}
	\operatorname{Newt}(p \odot q) = \operatorname{Newt}(p) + \operatorname{Newt}(q),
	\qquad
	\operatorname{Newt}(p \oplus q) = \operatorname{conv}\bigl( \operatorname{Newt}(p) \cup \operatorname{Newt}(q) \bigr).
	\label{eq:newton_calculus}
\end{equation}
Moreover, the lifted Newton polytope determines the polynomial function:
\begin{equation}

	\qquad \text{for all } x \in \RR^{d},

\end{equation}
and if all coefficients vanish, $c_{\alpha} \equiv 0$, then $p = h_{\operatorname{Newt}(p)}$.\end{proposition}

\begin{theoremenv}[DC--monotone splitting under integrable curvature, \LeanResult{thm:smooth_dc_split}]\label{thm:smooth_dc_split}
Let $f\colon \RR \to \RR$ be differentiable with locally absolutely continuous derivative $f'$ and $f'' \in L^1(\RR)$. Then the limit $s_- \coloneqq \lim_{u\to-\infty} f'(u)$ exists and $f'$ is bounded. Write $P \coloneqq \max\{f'',0\}$ and $N \coloneqq \max\{-f'',0\}$, so that $f'' = P - N$ with $P, N \ge 0$, and define
\begin{align}
	C_1(x) &\coloneqq f(0) + \int_0^x \Bigl( s_- + \int_{-\infty}^t P(u)\,\diff u \Bigr) \diff t,
	&
	C_2(x) &\coloneqq \int_0^x \int_{-\infty}^t N(u)\,\diff u \,\diff t,
	\label{eq:curvature_antiderivatives}
\end{align}
as well as $M \coloneqq \max\bigl\{ \sup_{x\in\RR} \abs{C_1'(x)},\, \sup_{x\in\RR} \abs{C_2'(x)} \bigr\} < \infty$. Then the functions
\begin{equation}
	g(a,b) \coloneqq C_1(a-b) + M(a+b),
	\qquad
	h(a,b) \coloneqq C_2(a-b) + M(a+b)
	\label{eq:smooth_split_pair}
\end{equation}
are convex on $\RR^2$, non-decreasing in each argument, and satisfy
\begin{equation}
	f(a-b) = g(a,b) - h(a,b) \qquad \text{for all } (a,b) \in \RR^2.
	\label{eq:smooth_splitting_identity}
\end{equation}\end{theoremenv}

\begin{proposition}[Minimality of the curvature split, \LeanResult{prop:curvature_minimality}]\label{prop:curvature_minimality}
Let $f$ be as in Theorem~\ref{thm:smooth_dc_split} and suppose $f = c_1 - c_2$ for convex functions $c_1, c_2 \colon \RR \to \RR$ whose second distributional derivatives are locally integrable functions. \begin{corollary}[Modern activation functions, \LeanResult{cor:modern_activations}]\label{cor:modern_activations}
Theorem~\ref{thm:smooth_dc_split} applies to the following activation functions, where $\varphi$ and $\Phi$ denote the standard Gaussian density and distribution function and $\sigma(x) = (1+e^{-x})^{-1}$ the logistic sigmoid:
\begin{enumerate}
	\item \emph{GELU}, $f(x) = x\,\Phi(x)$, with $f''(x) = (2 - x^2)\,\varphi(x)$; the curvature is positive on $(-\sqrt{2}, \sqrt{2})$ and negative outside.
	\item \emph{Swish/SiLU}, $f(x) = x\,\sigma(\beta x)$ with $\beta > 0$, with $f''(x) = \beta\,\sigma'(\beta x)\bigl( 2 + \beta x\,(1 - 2\sigma(\beta x)) \bigr)$; the curvature is positive on a bounded interval around the origin and negative in both tails.
	\item \emph{Mish}, $f(x) = x \tanh\bigl(\log(1 + e^{x})\bigr)$.
	\item \emph{Hyperbolic tangent}, $f(x) = \tanh(x)$, with $f''(x) = -2 \tanh(x)\operatorname{sech}^2(x)$, so $P$ is supported on $(-\infty, 0]$ and $N$ on $[0, \infty)$.
	\item \emph{Logistic sigmoid}, $f = \sigma$, with $f'' = \sigma'\,(1 - 2\sigma)$.
	\item \emph{Softplus at temperature $\theta > 0$}, $f(x) = \theta \log(1 + e^{x/\theta})$, with $f''(x) = \sigma'(x/\theta)/\theta \ge 0$; here $N = 0$, $C_2 = 0$, and the decomposition degenerates to the statement that softplus itself is convex.

\begin{definition}[Decomposition]\label{def:decomposition}
A \emph{decomposition} of a feedforward network~\eqref{eq:original_forward_pass} with $L$ layers is a calculation scheme that propagates non-negative activation pairs $\bigl(\act{l}{+}_i, \act{l}{-}_i\bigr) \in \RR^2_{\ge 0}$ from input to output using only
\begin{enumerate}
	\item \textbf{non-negative linear maps}---matrices with non-negative entries together with non-negative bias vectors---and
	\item \textbf{pairwise activation functions} $\phi \colon \RR^2_{\ge 0} \to \RR^2_{\ge 0}$, $(z^+, z^-) \mapsto (a^+, a^-) = \phi(z^+, z^-)$, applied at every hidden neuron; the final affine readout uses the identity on the pair.
\end{enumerate}
Activations are indexed by $l \in \{0,\dots,L\}$, affine parameters by $l \in \{1,\dots,L\}$. The layer-$0$ pair is any non-negative split of the input,
\begin{equation}
	\bigl(\act{0}{+}_k, \act{0}{-}_k\bigr) \coloneqq \bigl(x_k^+, x_k^-\bigr),
	\qquad
	x_k^+ - x_k^- = x_k,
	\qquad
	x_k^+, x_k^- \ge 0,
	\label{eq:input_split_convention}
\end{equation}
and at each affine layer $l$ a non-negative linear map sends the layer-$(l-1)$ activation pair to the layer-$l$ pre-activation pair $\bigl(\preact{l}{+}_i, \preact{l}{-}_i\bigr) \in \RR^2_{\ge 0}$, to which the pairwise activation is applied for $l<L$. At $l=L$ we set $\act{L}{\pm}=\preact{L}{\pm}$, with no activation or maximum. We call $\bigl(\act{l}{+}_i, \act{l}{-}_i\bigr)$ the \emph{decomposed activation} and $\bigl(\preact{l}{+}_i, \preact{l}{-}_i\bigr)$ the \emph{decomposed pre-activation}.
\end{definition}

\begin{remark}[On the $\eta = 0$ extreme]\label{rem:eta_zero_extreme}
The dual \ac{dc} representation writes the output as $u(x) - v(x)$ with $v$ \emph{concave}, and $\fmix[0]$ places $a_{\mathrm{mix},i}^{(l,-,0)} = \min\bigl\{\preact{l}{+}_i, \preact{l}{-}_i\bigr\}$ in the role of $v$. \begin{remark}[Preservation of sub-threshold signals]\label{rem:subthreshold}
The correctness theorem hides an information-theoretic asymmetry between the original and the split network. When a neuron is inactive, $\preactl{l}_i \le 0$, the original \ac{relu} outputs $\actl{l}_i = 0$ and discards the \emph{magnitude} of the losing pre-activation: downstream computation is oblivious to how strongly the neuron was suppressed. The split pair reads $\bigl(\act{l}{+}_i, \act{l}{-}_i\bigr) = \bigl(\preact{l}{-}_i, \preact{l}{-}_i\bigr)$: the difference is zero, as validity demands, but both components retain the accumulated magnitude $\preact{l}{-}_i$ of the cancelled mass and propagate it forward. The split streams thus carry strictly more information than the original activations---sub-threshold contributions survive the gate---which the attribution methods of Chapter~\ref{ch:saliency} exploit. The same effect drives the numerical growth of Section~\ref{sec:stabilization_dc}: without cancellation by subtraction, both streams accumulate the absolute mass $\preact{l}{+}_i + \preact{l}{-}_i$ instead of the signed residual.
\end{remark}

\begin{remark}[Shared gate versus per-stream maxima]\label{rem:gate_vs_maxima}
The contrast with the mixing family is sharpest at an inactive neuron, $\preactl{l}_i \le 0$. \end{remark}

\end{equation}

\begin{remark}[Probit regularizer and GELU]\label{rem:probit_gelu}
The framework is not tied to the entropy. For the probit regularizer $\Omega_\Phi(\pi) \coloneqq -\phi\bigl(\Phi^{-1}(\pi)\bigr)$, with $\phi, \Phi$ the standard Gaussian density and distribution function, one has $\frac{\diff}{\diff \pi}\bigl(-\Omega_\Phi\bigr)(\pi) = -\Phi^{-1}(\pi)$, so the first-order condition for~\eqref{eq:regularized_gate} gives the gate $\pi_{\Omega_\Phi}(\Delta) = \Phi(\Delta)$ and the conjugate activation
\begin{equation*}
	\Omega_\Phi^*(\Delta) = \Delta\, \Phi(\Delta) + \phi(\Delta) = \E_{\varepsilon \sim \mathcal{N}(0,1)}\bigl[ \max(\Delta + \varepsilon, 0) \bigr],
\end{equation*}
the Gaussian smoothing of the \ac{relu}. Up to $\phi(\Delta)$, which common implementations drop, this is GELU \parencite{Hendrycks2016}, and the gate $\Phi(\Delta) = \E_\varepsilon\bigl[\1[\Delta + \varepsilon > 0]\bigr]$ is the hard stopping gate averaged over a Gaussian shift of its threshold---matching the original stochastic-gate motivation for GELU. The distinction matters for exact decomposability: the conjugate activation $\Omega_\Phi^*(\Delta) = \Delta\,\Phi(\Delta) + \phi(\Delta)$ is the object recovered by $\freg$ and, being a Fenchel conjugate, is convex, non-decreasing, and $1$-Lipschitz (Theorem~\ref{thm:freg_validity}). The \emph{implemented} GELU $\Delta \mapsto \Delta\,\Phi(\Delta)$ that drops $\phi(\Delta)$ is non-monotone---it dips below zero for $\Delta < 0$---and is therefore \emph{not} itself a conjugate activation $\Omega^*$; the exactly decomposable object is the $+\phi(\Delta)$ variant, and networks using the dropped-term GELU are only approximately captured by this decomposition.
\end{remark}

\begin{remark}[Interpretation of the correction terms]\label{rem:Q_network}
The correction summands in~\eqref{eq:closed_formula_g} are gradients of an auxiliary network: define $Q[j](x) \coloneqq W_{\mathrm{abs}}^{[j,\rightarrow]}\min\{\preactl{j}(x),0\}$, with the original prefix through layer $j$ followed by a linear absolute-weight tail without biases or activation gates. In a bias-free tail this agrees in value with subsequent minimum activations, but the explicit definition also handles arbitrary prefix biases. Then the $j$-th summand equals $\overline{\alpha}_{j+1}\, \partial Q[j+1] / \partial x$ (at $l = 0$), capturing the mass that switches streams at layer $j+1$. These gradients grow rapidly toward the input, so the factors $\overline{\alpha}_j$ act as a per-layer lever that both prevents gradient explosion and regulates how strongly each layer's stream-switching enters the stabilized gradients. In practice all $\alpha^{(l)}$ equal a common $\alpha$; for $\alpha = 1/2$ the sensitivities are exact halves of the original gradient, for $\alpha = 0$ the raw split-pair gradients, and $0 < \alpha < 1/2$ is the interesting regime. These stabilized gradients are the backbone of the split saliency methods in Chapter~\ref{ch:saliency}.
\end{remark}

\begin{definition}[Payoffs, Transitions, and Value Recursion]\label{ptnb:def:adf-recursion}
Fix an input $x$.

\textbf{Payoffs.} The player-specific \emph{payoff} $r_p$ at activation states is paid out
depending on the action chosen by the active player, and paid out at terminal and cemetery states
without prerequisites:
\begin{itemize}
    \item Cemetery: $r_p(\dagger) \coloneqq 0$.
    \item Active state: $r_p(s_{j,q}^{(l, \mathrm{act})}, \mathrm{stop}) \coloneqq 0$ (stopping pays nothing)
        and $r_p(s_{j,q}^{(l, \mathrm{act})}, \mathrm{cont}) \coloneqq b_j^{(l, p \oplus q)}$ (continuation
        pays a positive bias to the active player $q$---covered by the case $p \oplus q = +$---and the
        absolute value of a negative bias to its opponent $q'$---covered by the case $p \oplus q = -$).
    \item Terminal: $r_p(s_{k,q}^{(0)}) \coloneqq x_k^{p \oplus q}$ (at terminal states the payoff depends on
        state only---the game ends here. A positive input is paid to the active player $q$---covered by the
        case $p \oplus q = +$---whereas the absolute value of a negative input is paid to its opponent
        $q'$---covered by the case $p \oplus q = -$).
\end{itemize}

\textbf{Transition kernel.} Stopping at state $s^{(l, \text{act})}_{j,p}$ deterministically sends
the game to the cemetery; continuation transports backward by one layer with transition
probabilities depending on the fraction of the respective absolute network weight compared to the
sum of absolute incoming weights at neuron $j$ of layer $l$. In case of a positive weight, the
active player $p$ stays in turn, in case of a negative weight, the turn switches to its opponent
$p'$.
\begin{align}\label{ptnb:eq:adf-kernel-stop}
    \P\bigl(\dagger \,\big|\, s_{j,q}^{(l, \mathrm{act})}, \mathrm{stop}\bigr) \coloneqq 1,
\end{align}
\begin{align}\label{ptnb:eq:adf-kernel}
    \P\bigl(s_{i,r}^{(l-1, \mathrm{act})} \,\big|\, s_{j,q}^{(l, \mathrm{act})}, \mathrm{cont}\bigr)
    \coloneqq \frac{W_{ij}^{(l, q \oplus r)}}{\gamma_j^{(l)}},
    \qquad
    \gamma_j^{(l)} \coloneqq \sum_i \bigl(W_{ij}^{(l,+)} + W_{ij}^{(l,-)}\bigr) = \sum_i
    \bigl|W_{ij}^{(l)}\bigr|,
\end{align}
where $\gamma_j^{(l)}$ is the local structural discount that restores unit row-sum (if
$\gamma_j^{(l)} = 0$, continuation leads to $\dagger$). For understanding the idea of the
occupation measure, we remark that those $\gamma^{(l)}_j$ are going to weight the trajectories in
the occupation measure.

\textbf{Policy.} A pure, i.e., deterministic policy $\pi = (\pi_+, \pi_-)$ assigns each active
state $s_{j,q}^{(l, \mathrm{act})}$ a binary stop/continue decision by the player in turn,
\begin{align}
    \pi_q\bigl(s_{j,q}^{(l, \mathrm{act})}\bigr) \in \{0, 1\},
    \qquad
    1 = \mathrm{cont},\ 0 = \mathrm{stop}.
\end{align}
\textbf{Player-specific action and game values.} For each policy $\pi$, the player-specific
\textbf{action value} $\tilde{Q}_p^\pi(s, a)$ is the expected discounted payoff for player $p$
starting at $s$ with action $a$ and following $\pi$ thereafter, defined by the standard Bellman
equation
\begin{align}\label{ptnb:eq:adf-q-bellman}
    \tilde{Q}_p^\pi(s, a) \coloneqq r_p(s, a) + \gamma(s)\, \textstyle\sum_{s'} \P\bigl(s' \,\big|\, s,
    a\bigr)\, \tilde{V}_p^\pi(s'),
\end{align}
where $\tilde{V}_p^\pi$ is the player-specific game value defined below. We use tildes to reserve
$V_p^\pi$ for the advantage further down, which already carries a subtracted $\tilde{V}_{p'}^\pi$.
At active states~\eqref{ptnb:eq:adf-q-bellman} specializes,
using~\eqref{ptnb:eq:adf-kernel-stop}--\eqref{ptnb:eq:adf-kernel}, to
\begin{align}\label{ptnb:eq:adf-q-stop}
    \tilde{Q}_p^\pi\bigl(s_{j,q}^{(l, \mathrm{act})},\, \mathrm{stop}\bigr) &= 0 + 1 \cdot
    \tilde{V}_p^\pi(\dagger) = 0 \\ \label{ptnb:eq:adf-q-cont}
    \tilde{Q}_p^\pi\bigl(s_{j,q}^{(l, \mathrm{act})},\, \mathrm{cont}\bigr)
    &= b_j^{(l, p \oplus q)} + \gamma_j^{(l)} \textstyle\sum_{s'} \P\bigl(s' \,\big|\, s_{j,q}^{(l,
    \mathrm{act})}, \mathrm{cont}\bigr)\, \tilde{V}_p^\pi(s') \\
    &= b_j^{(l, p \oplus q)} + \sum_{i,r} W_{ij}^{(l, q \oplus r)}\,
    \tilde{V}_p^\pi\bigl(s_{i,r}^{(l-1, \mathrm{act})}\bigr).
\end{align}
The \textbf{player-specific game value} $\tilde{V}_p^\pi(s)$ is the action value averaged by the
policy. At terminal and cemetery states it equals the local payoff there:
\begin{align}\label{ptnb:eq:adf-v-terminal}
    \tilde{V}_p^\pi\bigl(s_{k,q}^{(0)}\bigr) \coloneqq x_k^{p \oplus q},
    \qquad
    \tilde{V}_p^\pi(\dagger) \coloneqq 0.
\end{align}
At active states,
\begin{align}\label{ptnb:eq:adf-v-active}
    \tilde{V}_p^\pi\bigl(s_{j,q}^{(l, \mathrm{act})}\bigr) \coloneqq \pi_q\bigl(s_{j,q}^{(l,
    \mathrm{act})}\bigr)\, \tilde{Q}_p^\pi\bigl(s_{j,q}^{(l, \mathrm{act})},\, \mathrm{cont}\bigr) +
    \bigl(1 - \pi_q\bigl(s_{j,q}^{(l, \mathrm{act})}\bigr)\bigr)\, \tilde{Q}_p^\pi\bigl(s_{j,q}^{(l,
    \mathrm{act})},\, \mathrm{stop}\bigr).
\end{align}
Since $\pi_q$ is binary,~\eqref{ptnb:eq:adf-v-active} selects either $\tilde{Q}_p^\pi(\cdot,
    \mathrm{cont})$ or $\tilde{Q}_p^\pi(\cdot, \mathrm{stop}) = 0$.

\textbf{Game value.} The quantity the player in turn maximizes at each active state is the
advantage of player $p$ at $s$,
\begin{align}\label{ptnb:eq:adf-advantage}
    V_p^\pi(s) \coloneqq \tilde{V}_p^\pi(s) - \tilde{V}_{p'}^\pi(s).
\end{align}
\end{definition}

of game states
\begin{align}

layer-$0$ terminal state $\zeta_T = s_{k,r}^{(0)}$ (reaching an input dimension $k$) or at the
cemetery state $\zeta_T = \dagger$ (reached the first time a player chooses \emph{stop}). The set
of all such trajectories is the trajectory \emph{ground set} $\mathcal{T}_s$. Composing the
transition kernel~\eqref{ptnb:eq:adf-kernel-stop}--\eqref{ptnb:eq:adf-kernel}, the Oracle splits, and a
(possibly stochastic) policy $\pi$ at every activation state defines a Markov kernel on game

continuing past a closed gate carries probability zero. We write $\Pi_x(\cdot \mid s) \coloneqq

    \Pi_x(\cdot \mid s_0)$ for the law rooted at the output state $s_0 \coloneqq
    s_{u,+}^{(L,\mathrm{act})}$.

\begin{definition}[Entropy-Regularized Two-Player Routing Game]\label{ptnb:def:routing-recursion}
Fix an input $x$ and a stabilizer $\varepsilon \ge 0$. At each activation state $s_{j,p}^{(l,
    act)}$, the player in turn chooses \emph{stop} or \emph{continue} with similar mechanics as in the
Stopping Game (Section~\ref{ptnb:app:relu-net-game}) reflecting the open-closed state of a ReLU gate:
stopping sends the game to the cemetery $\dagger$ giving immediate payoff $\ell(0) = -\infty$---and
continuation transitions backward with structural discount $\gamma=2$ and the uniform $1/2$
sign-branch kernel given above. Here happens the critical step: gains of the opponent on the
negative stream state $s_{j,p',-}^{(l, \text{lin})}$ shall be counted as losses of the active
player $p$. At each linear routing state $s_{j,q,\kappa}^{(l, lin)}$, the player in turn instead
chooses a mixed action $\nu \in \Delta(\mathrm{Pred}_j^{(l)} \cup \{\mathrm{bias},
    \perp_{\mathrm{lin}}\})$ over the predecessors of neuron $j$ augmented by a terminal \emph{bias
node} with immediate payoff $\ell([b_j^{(l)}]^\kappa)$ and a fictitious \emph{outside option}
$\perp_{\mathrm{lin}}$ which routes the game's mass to the cemetery with immediate payoff $\ell
    (\varepsilon)$; trajectories entering either are absorbed. Let $\mathcal{H}(\nu) \coloneqq
    -\sum_{a} \nu(a)\log \nu(a)$ be Shannon entropy over this augmented action set. At $\varepsilon =
    0$ the outside option carries $\ell (0) = -\infty$ and is dominated by every positive-mass
predecessor, so the augmented game reduces to the Routing Game over $\mathrm{Pred}_j^{(l)} \cup
    \{\mathrm{bias}\}$.

An admissible profile chooses a pure stop/continue action at each activation state and a probability vector at each linear routing state. Under any profile, evaluate the displayed recursions using those choices, without taking maxima: use the chosen activation action and the chosen routing vector's expected log score plus entropy. A zero-probability term contributes zero even when its score is $-\infty$; positive probability on a $-\infty$ score makes the routing value $-\infty$.

Throughout, $\mathbf{V}(\cdot) = (V_p(\cdot), V_{p'}(\cdot))^\top$ denotes the game's
backward-induction player-specific log-value function, defined jointly at linear routing states
by~\eqref{ptnb:eq:routing-linear-value} and at activation states by~\eqref{ptnb:eq:routing-activation-value}
below. The recursion is well-posed because the layered state graph is a finite DAG: the $l=1$ case
of~\eqref{ptnb:eq:routing-linear-action} needs no lower-layer $\mathbf{V}$, and every subsequent layer
uses only $\mathbf{V}$ at strictly lower layers.

For a \textbf{linear routing state} at $l \geq 2$ the active player continuing to the $i$th
neuron's activation state receives an immediate payoff of the logarithm of the respective weight
summed with the respective log-value function of the next state, whereas in the first layer the
final payoff is paid out to the active player. The inactive player always obtains an immediate
payoff of $\ell(0) = -\infty$. Let us denote by $Q_r(s, a)$ player $r$'s gain for the game value in
case action $a$ actually happens, whereas $\bar Q_r$ denotes the mixture of the differences between
the payoffs of the opponents including the entropy regularization term given the mixed strategy
chosen by the active player.
\begin{align}\label{ptnb:eq:routing-linear-action}

    \coloneqq
    \begin{cases}
        \ell\!\bigl(W_{ij}^{(l,\kappa)}\bigr) + V_r\bigl(s_{i,q}^{(l-1, act)}\bigr), & r=q,\ l \ge 2,\\[0.35em]
        \ell\!\bigl([x_iW_{ij}^{(1)}]^\kappa\bigr), & r=q,\ l = 1,\\[0.35em]
        -\infty, & r=q'.
    \end{cases}
\end{align}
The outside option $\perp_{\mathrm{lin}}$ and the bias node have log-scores
$Q_r(s_{j,q,\kappa}^{(l,\mathrm{lin})}, \perp_{\mathrm{lin}}) \coloneqq \ell(\varepsilon)$ and
$Q_r(s_{j,q,\kappa}^{(l,\mathrm{lin})}, \mathrm{bias}) \coloneqq \ell([b_j^{(l)}]^\kappa)$ for $r =
    q$ and $-\infty$ for $r = q'$; they route the $\varepsilon$-fraction and the bias share of the mass
to absorbing terminals instead of to any predecessor.
If player $q$ uses mixed action $\nu$ at this state, the induced \emph{individual routed masses}
are
\begin{align}\label{ptnb:eq:optimal-linear}

    &\coloneqq
    \sum_a \nu(a)\, \ell \left( \exp \left( Q_r \bigl( s_{j,q,\kappa}^{(l, lin)}, a \bigr) \right) -
    \exp \left( Q_{r'} \bigl( s_{j,q,\kappa}^{(l, lin)}, a \bigr) \right) \right) \notag\\
    &\quad + \tau \mathcal{H}(\nu),
    \qquad r \in \{q,q'\},
\end{align}
which, as $\exp(-\infty)=0$, evaluates to
\begin{align}
    \bar Q_q \bigl( s_{j,q,\kappa}^{(l, lin)};\nu \bigr)
    &=
    \sum_a \nu(a)\, Q_q \bigl( s_{j,q,\kappa}^{(l, lin)}, a \bigr)
    + \tau \mathcal{H}(\nu) \\
    \bar Q_{q'} \bigl( s_{j,q,\kappa}^{(l, lin)};\nu \bigr)
    &= -\infty.
\end{align}
If $W_{ij}^{(l,\kappa)} = 0$ for every predecessor $i$ and $[b_j^{(l)}]^\kappa = 0$ (i.e.\ the
stream-$\kappa$ part of the fan-in to neuron $j$ is identically zero), then
$Q_q(s_{j,q,\kappa}^{(l,\mathrm{lin})},i) = -\infty$ for all $i$ as well and the
value~\eqref{ptnb:eq:routing-linear-value} equals $\ell(\varepsilon)$. If $\varepsilon=0$ and every
action scores $-\infty$ (e.g.\ $\Sigma_j^{(l,\kappa)}=0$ at $\tau=1$), every $\nu$ is a best
response; we fix $\nu=\delta_{\perp_{\mathrm{lin}}}$ (absorption), matching the convention
$0/0\coloneqq0$ in~\eqref{ptnb:eq:lrp-eps-recursion}.

\begin{align}\label{ptnb:eq:routing-linear-value}

    &\coloneqq
    \max_{\nu \in \Delta(\mathrm{Pred}_j^{(l)} \cup \{\mathrm{bias}, \perp_{\mathrm{lin}}\})}
    \bar Q_q\bigl(s_{j,q,\kappa}^{(l, lin)};\nu\bigr) \nonumber \\

    + \tau \mathcal{H}(\nu)
    \right\} \nonumber \\

    \bigl([b_j^{(l)}]^\kappa\bigr)^{\frac{1}{\tau}} + \sum_i \exp\left( Q_q(s_{j,q,\kappa}^{(l, lin)},
    i) \right)^{ \frac{ 1 }{ \tau } } \right), \\
    V^\star_{q'} \bigl( s_{j,q,\kappa}^{(l, lin)} \bigr)
    &\coloneqq -\infty,
\end{align}
where we used~\eqref{ptnb:eq:gibbs-lse}. At $\tau = 1$ the associated Gibbs policy assigns the outside
option the probability $\varepsilon / (\varepsilon + \sum_{a \neq \perp_{\mathrm{lin}}} \exp
    Q_q(\cdots, a))$ and distributes the remaining mass over the predecessors and the bias node in
proportion to $\exp Q_q$.

For an \textbf{activation state}, continuing exposes both sign branches. Let $s_+ \coloneqq
    s_{j,p,+}^{(l, \text{lin})}$ and $s_- \coloneqq s_{j,p',-}^{(l, \text{lin})}$. If player $p$
chooses \emph{continue} at $s_{j,p}^{(l, act)}$, the game transitions to $s_+$ or to $s_-$ with
equal probability, while \emph{stop} sends the game to the cemetery with game value
$V_r(\dagger)=\ell(0)=-\infty$. Taking the expectation over the uniform sign-branch kernel

    &\coloneqq
    \ell \left( \sum_{s' \in \{s_+, s_-\}} \P(s' \mid s_{j,p}^{(l, act)}) \cdot \gamma_{\mathrm{O}}
    \cdot \Bigl( \exp V_r(s') - \exp V_{r'}(s') \Bigr) \right) \nonumber \\
    &= \ell \left(
    \xi_{r,p} \Bigl( \exp\left( V_p\left( s_+ \right) \right) - \exp\left( V_{p'}\left( s_- \right)
    \right) \Bigr)
    \right).
\end{align}

\begin{align}\label{ptnb:eq:routing-activation-value}

    &\coloneqq \1\!\left[Q_p(s_{j,p}^{(l,\mathrm{act})},\mathrm{continue})>-\infty\right],\\
    V_p(s_{j,p}^{(l,\mathrm{act})})
    &\coloneqq \max\{-\infty,Q_p(s_{j,p}^{(l,\mathrm{act})},\mathrm{continue})\}.
\end{align}
Writing $D_p=\exp V_p(s_+)-\exp V_{p'}(s_-)$, this selects continuing exactly when $D_p>0$. When $D_p\leq0$, both actions have log value $-\infty$ and the convention selects stopping. If continuing is selected, the opponent's continuation value is $-\infty$; if stopping is selected, both players receive $-\infty$. Thus in either case
\begin{align}

\begin{definition}[Maxout multiplication network]\label{def:maxout_multiplication_network}
A \emph{rank-$k$ Maxout multiplication network} is a finite directed acyclic graph of \emph{input neurons}, \emph{addition neurons}, and \emph{multiplication neurons}. An addition neuron $v$ carries weight vectors $w^{uv} \in \RR^k$ on its incoming arcs and computes the rank-$k$ Maxout activation $a^v = \max_{i \in [k]} \sum_{u} w^{uv}_i a^u$; a multiplication neuron carries natural exponents $w^{uv} \in \NN$ and computes the monomial $a^v = \prod_{u} (a^u)^{w^{uv}}$. The network is called \emph{monotone} if all weights are non-negative, and \emph{input-convex} if all weights are non-negative except possibly those on arcs leaving input neurons.
\end{definition}

\end{equation*}
Each component is monotone as a function of its non-negative split inputs and is representable there by a finite maximum of posynomials. \begin{remark}[A probabilistic reading]\label{rem:splitlrp_markov}
On the original network the $\varepsilon$-rule mixes signed contributions, and the resulting backward transport is a \emph{signed} measure on paths --- the source of the interpretability caveats discussed in Chapter~\ref{ch:trajectory}. \begin{remark}[Routing-state elimination and absorption histories, \LeanResult{ptnb:rem:rg-natural-equivalence}]\label{ptnb:rem:rg-natural-equivalence}
The refined diagnostic routing process just defined alternates activation and linear-routing states.
A dead activation stops at the gate substep; an alive activation chooses its
$\kappa$ branch uniformly, after which an empty branch or outside option can
absorb at the routing substep. The refined layered process retains these
substeps and pads the cemetery state after absorption. Algorithm~\ref{ptnb:alg:hellinger}
applies to this expanded state graph; the diagnostics are read at the activation
boundaries. This is the process implemented by the separate gate and branch
absorption updates.

For a trajectory that remains ordinary, deleting each intermediate routing
state is a bijection: its neuron and player are determined by the adjacent
activation states, and its branch by whether the player is kept or switched
(or by the pixel-stream tag at the input). The act-to-act formulas therefore
preserve ordinary surviving-path probabilities and their Bhattacharyya overlap.
The same applies after conditioning both laws on survival.

For absorbing trajectories, deletion need not be injective: a gate stop and a
later empty-branch stop can become the same act-only trajectory. Thus full-law
Bhattacharyya and Hellinger equality is asserted only when absorption histories
are recoverable, for example when no alive routing branch leaks to the cemetery.
In general the finite-fibre Cauchy--Schwarz inequality gives
\[
 \mathrm{BC}_{\mathrm{refined}}\leq\mathrm{BC}_{\mathrm{act\text{-}only}},
 \qquad
 \mathrm{H}_{\mathrm{refined}}\geq\mathrm{H}_{\mathrm{act\text{-}only}}.
\]
For example, take one input $x=1$ and weight $W=1$. Model $A$ has bias $0$ and
continues, with its empty negative branch absorbing with probability $1/2$;
model $B$ has bias $-2$ and stops at the gate. Their refined path supports are
disjoint, so $\mathrm{BC}_{\mathrm{refined}}=0$. Collapsing the two absorption
substeps instead gives $\mathrm{BC}_{\mathrm{act\text{-}only}}=1/\sqrt2$.\end{remark}

    \mathrm{H}^2(s) \;\coloneqq\; \tfrac{1}{2}\bigl(m^{(0)}_A(s) + m^{(0)}_B(s)\bigr) \;-\; g^{(0)}(s),
\end{align}
where $g^{(0)}(s)$ is the cumulative $\widetilde{T}$-mass at $s$.  Then:
\begin{enumerate}
    \item[(i)] $\mathrm{H}^2(s) \geq 0$ for every $s \in \mathcal{S}^{(0)}_+$.
    \item[(ii)]
        \begin{align}\label{ptnb:eq:per-pixel-sum}
            \mathrm{H}^2 \;=\; \sum_{s \in \mathcal{S}^{(0)}_+} \mathrm{H}^2(s).
        \end{align}
    \item[(iii)]
        \begin{align}\label{ptnb:eq:ordinary-h2}
            \mathrm{H}^2_{\mathrm{ord}} \;\coloneqq\; \sum_{s \in \mathcal{S}^{(0)}} \mathrm{H}^2(s) \;=\;
            \mathrm{H}^2 - \mathrm{H}^2(\dagger)

\begin{theoremenv}[Conditioned-survival trajectory law and conditioned Hellinger distance, \LeanResult{ptnb:thm:conditioned-survival}]\label{ptnb:thm:conditioned-survival}
Let $\mathsf{Surv} \coloneqq \{(\zeta_L,\ldots,\zeta_0) : \zeta_0 \in \mathcal{S}^{(0)}\}$ be the
event, in the trajectory space $\prod_{l=0}^{L} \mathcal{S}^{(l)}_+$, that the backward trajectory
ends in an ordinary input-layer state — equivalently, the event that the trajectory never enters
the cemetery~$\dagger$.  For each model $M \in \{A,B\}$ define the survival function
\begin{align}\label{ptnb:eq:survival-function}
    \psi_M^{(l)}(s)
    \;\coloneqq\;
    \mathbb{P}_M(\mathsf{Surv} \mid \zeta_l = s),
    \qquad s \in \mathcal{S}^{(l)}_+, \;\; l \in \{0,\ldots,L\}.
\end{align}
Then:
\begin{enumerate}
    \item[(i)] The boundary and recursion are
        \begin{align}\label{ptnb:eq:survival-boundary}
            \psi_M^{(0)}(s) = 1 \;\; \text{for } s \in \mathcal{S}^{(0)},
            \qquad
            \psi_M^{(0)}(\dagger) = 0,
        \end{align}
        and for every $l \ge 1$,
        \begin{align}\label{ptnb:eq:survival-recursion}
            \psi_M^{(l)}(s)
            \;=\;
            \sum_{s' \in \mathcal{S}^{(l-1)}_+} T_M^{(l)}(s \to s')\, \psi_M^{(l-1)}(s')
            \;=\;
            \sum_{s' \in \mathcal{S}^{(l-1)}} T_M^{(l)}(s \to s')\, \psi_M^{(l-1)}(s').
        \end{align}
        In particular,
        \begin{align}\label{ptnb:eq:survival-mass}
            Z_M \;\coloneqq\; \mathbb{P}_M(\mathsf{Surv}) \;=\; \psi_M^{(L)}(s_0) \;=\; \sum_{s \in
            \mathcal{S}^{(0)}} m_M^{(0)}(s) \;=\; 1 - m_M^{(0)}(\dagger).
        \end{align}
    \item[(ii)] Assume $Z_M > 0$.  For every ordinary state $s \in \mathcal{S}^{(l)}$ with $\psi_M^{(l)}(s) >
            0$, define the conditioned kernel on ordinary states by
        \begin{align}\label{ptnb:eq:conditioned-kernel}
            T_{M,\mathrm{surv}}^{(l)}(s \to s')
            \;\coloneqq\;
            \frac{T_M^{(l)}(s \to s')\, \psi_M^{(l-1)}(s')}{\psi_M^{(l)}(s)},
            \qquad s' \in \mathcal{S}^{(l-1)}.
        \end{align}
        Then $T_{M,\mathrm{surv}}^{(l)}(s \to \cdot)$ is row-stochastic on $\mathcal{S}^{(l-1)}$, and the
        conditional trajectory distribution $\Pi_{M,\mathrm{surv}} \coloneqq \Pi_M(\,\cdot \mid
            \mathsf{Surv}\,)$ factorizes as the layered Markov law
        \begin{align}\label{ptnb:eq:conditioned-path-measure}
            \Pi_{M,\mathrm{surv}}(\zeta_L,\ldots,\zeta_0)
            = \mathbf{1}\{\zeta_L = s_0\} \prod_{k=1}^{L} T_{M,\mathrm{surv}}^{(k)}(\zeta_k \to \zeta_{k-1}),
            \qquad (\zeta_L,\ldots,\zeta_0) \in \mathsf{Surv}.
        \end{align}
        Its layer-$l$ marginals satisfy
        \begin{align}\label{ptnb:eq:conditioned-alpha}
            m_{M,\mathrm{surv}}^{(l)}(s)
            \;=\;
            \frac{m_M^{(l)}(s)\, \psi_M^{(l)}(s)}{Z_M},
            \qquad s \in \mathcal{S}^{(l)}.
        \end{align}
    \item[(iii)] Define the conditioned geometric-mean kernel
        \begin{align}\label{ptnb:eq:conditioned-geom-kernel}
            \widetilde{T}_{\mathrm{surv}}^{(l)}(s,s')
            \coloneqq \sqrt{T_{A,\mathrm{surv}}^{(l)}(s \to s')\, T_{B,\mathrm{surv}}^{(l)}(s \to s')}
            = \frac{\widetilde{T}^{(l)}(s,s')\, \sqrt{\psi_A^{(l-1)}(s')\,
            \psi_B^{(l-1)}(s')}}{\sqrt{\psi_A^{(l)}(s)\, \psi_B^{(l)}(s)}}.
        \end{align}
        The Bhattacharyya coefficient of the conditioned trajectory laws is
        \begin{align}\label{ptnb:eq:conditioned-bc-kernel}
            \mathrm{BC}\!\bigl(\Pi_{A,\mathrm{surv}}, \Pi_{B,\mathrm{surv}}\bigr)
            \;=\;
            \sum_{(\zeta_L,\ldots,\zeta_0) \in \mathsf{Surv}}
            \mathbf{1}\{\zeta_L = s_0\}
            \prod_{k=1}^{L} \widetilde{T}_{\mathrm{surv}}^{(k)}(\zeta_k, \zeta_{k-1}),
        \end{align}
        so it is computed by the same Hellinger pass (Algorithm~\ref{ptnb:alg:hellinger}), but on the
        ordinary-state graph with the conditioned kernel.
    \item[(iv)] Let
        \begin{align}\label{ptnb:eq:ordinary-bc}
            \mathrm{BC}_{\mathrm{ord}}^{(0)}
            \;\coloneqq\;
            \sum_{s \in \mathcal{S}^{(0)}} g^{(0)}(s)
            \;=\;
            \sum_{(\zeta_L,\ldots,\zeta_0) \in \mathsf{Surv}} \sqrt{\Pi_A(\zeta_L,\ldots,\zeta_0)\,
            \Pi_B(\zeta_L,\ldots,\zeta_0)}.
        \end{align}
        Then
        \begin{align}\label{ptnb:eq:conditioned-bc-posthoc}
            \mathrm{BC}\!\bigl(\Pi_{A,\mathrm{surv}}, \Pi_{B,\mathrm{surv}}\bigr)
            \;=\;
            \frac{\mathrm{BC}_{\mathrm{ord}}^{(0)}}{\sqrt{Z_A Z_B}},
        \end{align}
        and the conditioned-survival Hellinger distance is
        \begin{align}\label{ptnb:eq:conditioned-h2}
            \mathrm{H}_{\mathrm{surv}}^2
            \;\coloneqq\;
            \mathrm{H}^2\!\bigl(\Pi_{A,\mathrm{surv}}, \Pi_{B,\mathrm{surv}}\bigr)
            \;=\;
            1 - \frac{\mathrm{BC}_{\mathrm{ord}}^{(0)}}{\sqrt{Z_A Z_B}}.
        \end{align}
\end{enumerate}\end{theoremenv}

\begin{remark}[Cemetery floor]\label{ptnb:rem:adf-cemetery-floor}
Theorem~\ref{ptnb:thm:rand-adf} bounds only the ordinary-state overlap.  Overlap mass that reaches a
neuron dead in both models enters the cemetery in both, $g^{(l-1)}(\dagger) = g^{(l)}(\dagger) +
    \sum_{s \in \mathrm{dead}^{(l)}_A \cap \mathrm{dead}^{(l)}_B} g^{(l)}(s)$, and is never thinned
again, so plain $\mathrm{H}$ saturates below one.  In a mean-field approximation with uniform dead
fraction $q$ under $A$ and alive probability $p$ under $B$, the ordinary overlap shrinks by
$(1-q)p$ per layer while a fraction $q(1-p)$ is absorbed, so $g^{(0)}(\dagger) \to
    q(1-p)/(1-(1-q)p)$ as $N \to \infty$; for illustrative values $q = 0.2$, $p = \tfrac{1}{2}$ this
gives $\mathrm{H} \approx 0.91$, the same order as the observed $\drFloorMin$--$\drFloorMax$
(Table~\ref{ptnb:tab:double-random-hellinger}).  The conditioned-survival distance removes the cemetery
floor exactly, and $\mathbb{E}_B[\mathrm{H}^2_{\mathrm{surv}} \mid Z_A Z_B \geq z] \to 1$ for every
fixed $z > 0$ with $\mathbb{P}_B(Z_A Z_B \geq z)$ bounded away from zero.  For the Routing Game the
activation-weighted kernel concentrates $g^{(l)}$ on neurons with largest $|a W|$, which rarely
coincide with $\mathrm{dead}_A \cap \mathrm{dead}_B$, so $\mathrm{H}$ is nearly identical to
$\mathrm{H}_{\mathrm{surv}}$ in practice.
\end{remark}
