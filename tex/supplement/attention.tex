\section{State space}\label{ptnb:app:attention-state-space}

We embed attention into the routing graph of \S\ref{ptnb:app:routing-formal}, writing
$s_\bullet^{(\mathrm{att},\,\mathrm{Side},\,\mathrm{type})}$, where
$\mathrm{Side} \in \{\mathrm{O}, \mathrm{val}, \mathrm{X}\}$ identifies the activation slot and
$\mathrm{type} \in \{\mathrm{act}, \mathrm{lin}\}$ the state type of \S\ref{ptnb:app:routing-formal}.
For AV mixing and the Value projection, $t,k \in \{1,\dots,S\}$ index output/key tokens,
$d \in \{1,\dots,d_h\}$ indexes features within a fixed head $h$, and
$e \in \{1,\dots,D\}$ indexes per-token input embedding dimensions.

\paragraph{States for the composite $Av{+}W_v$ block.} The explanatory walk adds three state types to the surrounding ReLU-network's,
all indexed by the player in turn.
\begin{itemize}
    \item \textbf{Output activation states} $s_{t,d,p}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{act})}$ at the
        head's output entry $O_{t,d}$, indexed by output token $t \in \{1,\dots,S\}$, within-head feature
        dim $d \in \{1,\dots,d_h\}$, and player $p \in \{+,-\}$. The surrounding linear-layer recursion of \S\ref{ptnb:app:routing-attribution}
        delivers occupation mass to these entry states $\{s_{t,d,p}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{act})}\}_{t,d,p}$
        from the layer behind the head.
    \item \textbf{Output sign-branch (linear) states}
        $s_{(t,d),p,\kappa}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}$, indexed by $(t, d, p, \kappa)$.
        These mirror the linear routing states $s_{j,q,\kappa}^{(l,\mathrm{lin})}$ of
        Definition~\ref{ptnb:def:routing-recursion} for AV mixing at $(t, d)$.
        QK logits enter through the reference $\bar\pi_t = A_{t,\cdot}$: the active
        player picks key $k$ using a mixed action $\pi_{t,d,\kappa}^\star \in \Delta(\{1,\dots,S\})$
        that combines the AV reference $\bar\pi_t$ with the $\kappa$-stream Value mass
        $\tilde{v}_{k,d}^\kappa$.
    \item \textbf{Value-projection linear states}
        $s_{(k,d),p,\kappa}^{(\mathrm{att},\,\mathrm{val},\,\mathrm{lin})}$, indexed by key token $k$,
        within-head feature dim $d$, player $p$, and sign $\kappa$. Here the player routes through the per-token Value projection to input dimension $e$
        (or the bias node), with game value
        $V_p\!\bigl(s_{(k,d),p,\kappa}^{(\mathrm{att},\,\mathrm{val},\,\mathrm{lin})}\bigr) =
            \ell(\tilde{v}_{k,d}^\kappa)$ with $\tilde{v}_{k,d}^\kappa$ from~\eqref{ptnb:eq:attn-Z-def}.
    \item \textbf{Input activation states} $s_{(k,e),p}^{(\mathrm{att},\,\mathrm{X},\,\mathrm{act})}$ at the
        input embedding entry $X_{k,e}$, indexed by token--embedding-dim pair $(k,e)$ and player $p$. These exit states lie upstream of the Value projection, indexed by embedding dimension $e$;
        the head-feature axis $d$ enters at its output. The surrounding recursion resumes from
        their per-player occupation on $\{s_{(k,e),p}^{(\mathrm{att},\,\mathrm{X},\,\mathrm{act})}\}_{(k,e),p}$.
\end{itemize}

\paragraph{Transitions.} The three-step trajectory has discount $1$ except at the sign-oracle split,
which uses $\{2\alpha, 2\beta\}$ as at ReLU activation states in
Definition~\ref{ptnb:def:routing-recursion}. Each output $(t,d)$ has \emph{one} sign-oracle split:
$A_{t,k} \ge 0$ absorbs AV mixing into the $W_v$ sign split.
\begin{enumerate}
    \item \emph{Sign-oracle split (output activation $\to$ output sign-branch, fixed $(t,d)$).} At
        $s_{t,d,p}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{act})}$ an unobserved Oracle transitions uniformly
        to $s_{(t,d),p,+}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}$ (player $p$ retains the turn,
        trajectory discount $2\alpha$) or to $s_{(t,d),p',-}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}$
        (turn switches to opponent $p'$, discount $2\beta$), each with probability $\tfrac12$. The feature
        index $d$ is preserved.
    \item \emph{Value-routing policy (output sign-branch $\to$ Value-projection linear).} At
        $s_{(t,d),p,\kappa}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}$ the active player picks a key
        token $k$ by the mixed action
        \begin{align}\label{ptnb:eq:attn-V-routing-policy-O}
            \pi_{t,d,\kappa}^\star(k)
            \;=\; \frac{A_{t,k}\, \tilde{v}_{k,d}^\kappa}{Z_{t,d}^\kappa},
        \end{align}
        derived in \S\ref{ptnb:app:attention-value-routing} as the equilibrium of a KL-regularized log-payoff
        problem against the reference $\bar\pi_t = A_{t,\cdot}$. The transition to
        $s_{(k,d),p,\kappa}^{(\mathrm{att},\,\mathrm{val},\,\mathrm{lin})}$ preserves player,
        sign $\kappa$ and feature $d$, changing only the token index $t$ to $k$, with discount $1$.
    \item \emph{Value-projection routing (Value-projection linear $\to$ input activation).} At
        $s_{(k,d),p,\kappa}^{(\mathrm{att},\,\mathrm{val},\,\mathrm{lin})}$ the active player picks an
        input dim $e$ by the standard linear-state Gibbs policy of Definition~\ref{ptnb:def:routing-recursion}
        on the $\kappa$-stream masses $[X_{k,e}W_{v,e,d}]^\kappa$ (the bias node receives
        $[b_{v,d}]^\kappa/\tilde{v}_{k,d}^\kappa$ and is absorbed):
        \begin{align}\label{ptnb:eq:attn-V-routing-policy}
            \pi_{\mathrm{val},k,d,\kappa}^\star(e)
            \;=\; \frac{[X_{k,e} W_{v,e,d}]^\kappa}{\tilde{v}_{k,d}^\kappa}.
        \end{align}
        The trajectory transitions to $s_{(k,e),p}^{(\mathrm{att},\,\mathrm{X},\,\mathrm{act})}$ with the
        player label preserved and trajectory-discount factor $1$.
\end{enumerate}
We evaluate the discounted occupation measure $\Gamma_x(\,\cdot\,)$ of
\S\ref{ptnb:app:routing-attribution} at all four state types.
The recovery theorem of \S\ref{ptnb:app:attention-recovery} compares its input-neuron
player-differences with the target rule~\eqref{ptnb:eq:attn-target-input}.

\section{Legendre--Fenchel transform of the KL divergence}\label{ptnb:app:kl-legendre-fenchel}

The Value-routing equilibrium follows from the KL convex-conjugate identity:

\begin{lemmaenv}[Legendre--Fenchel transform of KL divergence, \LeanResult{ptnb:lem:kl-legendre-fenchel}]\label{ptnb:lem:kl-legendre-fenchel}
Let $\mathcal{X}$ be a finite set, $\bar\pi \in \Delta(\mathcal{X})$ a reference distribution with
full support on $\mathcal{X}$, and $r : \mathcal{X} \to \RR$ a real-valued payoff. The
KL-regularized linear maximization
\begin{align}\label{ptnb:eq:kl-lf-objective}
    F(r;\bar\pi) \;\coloneqq\; \max_{\pi \in \Delta(\mathcal{X})}
    \Bigl\{ \E_\pi[r] \;-\; \mathrm{KL}(\pi \,\|\, \bar\pi) \Bigr\}
\end{align}
has optimizer and optimum:
\begin{align}\label{ptnb:eq:kl-lf-optimizer}
    \pi_r^\star(x) \;=\; \frac{\bar\pi(x)\,\exp\!\bigl(r(x)\bigr)}{Z(r;\bar\pi)},
    \qquad
    F(r;\bar\pi) \;=\; \log Z(r;\bar\pi) \;=\; \log\!\sum_{x \in \mathcal{X}}
    \bar\pi(x)\,\exp\!\bigl(r(x)\bigr).
\end{align}\end{lemmaenv}

Two specializations are used below. For $r(x) = \ell(u(x))$, with
$u : \mathcal{X} \to \RR_{\geq 0}$ and $\exp(-\infty) \coloneqq 0$, we have $\exp(r(x)) = u(x)$.
The value is the logarithm of the $\bar\pi$-weighted sum, and the optimizer its tilted normalization:
\begin{align}\label{ptnb:eq:kl-lf-linearized}
    F\bigl(\ell(u);\,\bar\pi\bigr) \;=\; \ell\!\Bigl(\sum_{x}\bar\pi(x)\,u(x)\Bigr),
    \qquad
    \pi_{\ell(u)}^\star(x) \;=\; \frac{\bar\pi(x)\,u(x)}{\sum_{x'}\bar\pi(x')\,u(x')}.
\end{align}
When $\bar\pi$ is uniform on $\mathcal{X}$, $\mathrm{KL}(\pi\,\|\,\mathrm{Unif}_\mathcal{X}) =
    \log|\mathcal{X}| - \mathcal{H}(\pi)$, so the KL penalty in~\eqref{ptnb:eq:kl-lf-objective}
becomes an entropy bonus:
\begin{align}
    \E_\pi[r] \;-\; \mathrm{KL}\!\bigl(\pi\,\|\,\mathrm{Unif}_\mathcal{X}\bigr)
    \;=\; \E_\pi[r] \;+\; \mathcal{H}(\pi) \;-\; \log|\mathcal{X}|.
\end{align}
Up to the additive constant $-\log|\mathcal{X}|$,~\eqref{ptnb:eq:kl-lf-objective} therefore reduces to
the entropy-regularized maximization of~\eqref{ptnb:eq:gibbs-lse}, with value $\log\sum_{x}\exp(r(x))$
and optimizer $\pi^\star(x) \propto \exp(r(x))$.

\begin{proof}
This is the Gibbs variational principle~\parencite{BoydVandenberghe2004}: for every $\pi$, $\E_\pi[r] -
    \mathrm{KL}(\pi\,\|\,\bar\pi) = \log Z(r;\bar\pi) - \mathrm{KL}(\pi\,\|\,\pi_r^\star)$, which is
maximal, with value $\log Z(r;\bar\pi)$, exactly at $\pi = \pi_r^\star$.
\end{proof}

\section{Value-routing equilibrium}\label{ptnb:app:attention-value-routing}

Fix an output entry $(t, d)$, a sign $\kappa \in \{+,-\}$, and the reference $\bar\pi_t =
    A_{t,\cdot}$ from \S\ref{ptnb:app:attention-state-space} (shared across feature dimensions $d$ by AV mixing). At the output sign-branch state
$s_{(t,d),p,\kappa}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}$ the active player $p$ solves the
KL-regularized log-payoff maximization
\begin{align}\label{ptnb:eq:attn-value-routing}
    V_p\!\bigl(s_{(t,d),p,\kappa}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}\bigr)
    \;\coloneqq\;
    \max_{\pi \in \Delta(\{1,\dots,S\})}
    \left\{
    \sum_{k=1}^S \pi(k)\, \ell\!\bigl(\tilde{v}_{k,d}^\kappa\bigr)
    \;-\;
    \mathrm{KL}\!\bigl(\pi \,\|\, \bar\pi_t\bigr)
    \right\},
\end{align}
where $\tilde{v}_{k,d}^\kappa \ge 0$ is the $\kappa$-stream split of the Value entry $v_{k,d}$
from~\eqref{ptnb:eq:attn-Z-def} and $\ell$ is the extended logarithm of \S\ref{ptnb:app:routing-formal}. The
payoff $\ell(\tilde{v}_{k,d}^\kappa)$ on action $k$ is the active-player value of the predecessor
Value-projection linear state,
$V_p(s_{(k,d),p,\kappa}^{(\mathrm{att},\,\mathrm{val},\,\mathrm{lin})}) =
    \ell(\tilde{v}_{k,d}^\kappa)$, so combining with the AV log-weight $\ell(A_{t,k})$ gives composite
payoff $\ell(A_{t,k}\, \tilde{v}_{k,d}^\kappa)$ at the $(t,d)$-output. Define the $\bar\pi_t$-weighted Value mass at each output:
\begin{align}\label{ptnb:eq:attn-z-def}
    Z_{t,d}^\kappa \;\coloneqq\; \sum_{m=1}^S \bar\pi_t(m)\, \tilde{v}_{m,d}^\kappa.
\end{align}
Only $\tilde{v}_{m,d}^\kappa$ depends on $d$.

\begin{proposition}[Value-side optimal policy, \LeanResult{ptnb:prop:attn-value-policy}]\label{ptnb:prop:attn-value-policy}
If $Z_{t,d}^\kappa > 0$, the unique optimizer of~\eqref{ptnb:eq:attn-value-routing} is
\begin{align}\label{ptnb:eq:attn-value-policy}
    \pi_{t,d,\kappa}^\star(k)
    \;=\;
    \frac{\bar\pi_t(k)\, \tilde{v}_{k,d}^\kappa}{Z_{t,d}^\kappa},
\end{align}
with active-player value $V_p(s_{(t,d),p,\kappa}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}) =
    \ell(Z_{t,d}^\kappa)$ and inactive-player value
$V_{p'}(s_{(t,d),p,\kappa}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}) = -\infty$ as in
Definition~\ref{ptnb:def:routing-recursion}. If $Z_{t,d}^\kappa = 0$, no key contributes mass on stream
$\kappa$ and $V_p(s_{(t,d),p,\kappa}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{lin})}) = -\infty$. All key policies then have value $-\infty$;
the maximizer is nonunique. We use the cemetery completion above.\end{proposition}

\begin{proof}
Apply~\eqref{ptnb:eq:kl-lf-linearized} of Lemma~\ref{ptnb:lem:kl-legendre-fenchel}
with $\mathcal{X} = \{1,\dots,S\}$, $\bar\pi = \bar\pi_t$ and $u(k) = \tilde{v}_{k,d}^\kappa$.
Then $r(k) = \ell(\tilde{v}_{k,d}^\kappa)$ gives $\exp r(k) = \tilde{v}_{k,d}^\kappa$,
yielding~\eqref{ptnb:eq:attn-value-policy} and value $\ell(Z_{t,d}^\kappa)$ for $Z_{t,d}^\kappa > 0$. The
inactive-player value $-\infty$ is the Routing-Game convention of
Definition~\ref{ptnb:def:routing-recursion}, inherited from the $r = q'$ branch
of~\eqref{ptnb:eq:routing-linear-action}. If $Z_{t,d}^\kappa = 0$ then
$\bar\pi_t(k)\,\tilde{v}_{k,d}^\kappa = 0$ for every $k$ and the supremum
of~\eqref{ptnb:eq:attn-value-routing} over the simplex is $-\infty$, consistent with $\ell(0) =
    -\infty$.
\end{proof}

\paragraph{Specialization to the softmax reference.}
Setting $\bar\pi_t = A_{t,\cdot}$ in~\eqref{ptnb:eq:attn-value-policy} yields the closed-form
per-$(t,d,\kappa)$ policies used in the \ac{rg}'s backward walk through the attention block,
\begin{align}\label{ptnb:eq:attn-branch-policy}
    \pi_{t,d,+}^\star(k) \;=\; \frac{A_{t,k}\, \tilde{v}_{k,d}^+}{Z_{t,d}^+},
    \qquad
    \pi_{t,d,-}^\star(k) \;=\; \frac{A_{t,k}\, \tilde{v}_{k,d}^-}{Z_{t,d}^-},
\end{align}
when $Z_{t,d}^\kappa>0$. When $Z_{t,d}^\kappa=0$, set all these key probabilities to zero
and the cemetery probability to one. Here $Z_{t,d}^\kappa = \sum_m A_{t,m}\, \tilde{v}_{m,d}^\kappa$
is defined in~\eqref{ptnb:eq:attn-z-def}. Using the cached softmax row $\bar\pi_t = A_{t,\cdot}$ as a fixed input
to~\eqref{ptnb:eq:attn-value-routing} suffices; the oracle interpretation of
\S\ref{ptnb:app:attention-oracle} is optional.

\section{\texorpdfstring{$\alpha\beta$-\ac{lrp}}{alpha-beta-LRP} recovery for the attention block}\label{ptnb:app:attention-recovery}

For the four-state trajectory of \S\ref{ptnb:app:attention-state-space}, discounted
by~\eqref{ptnb:eq:routing-edge-discount}, the player-difference of $\Gamma_x$ at the
\emph{input activation states} recovers the target rule~\eqref{ptnb:eq:attn-target-input} of
\S\ref{ptnb:app:attention-target-rule}. These states precede the Value-projection linear states,
where $W_v$ acts. The two objects are:
\begin{itemize}
    \item the \emph{game-side} object is the discounted occupation measure $\Gamma_x(\,\cdot\,)$ of
        \S\ref{ptnb:app:routing-attribution} evaluated at the four attention-block state types of
        \S\ref{ptnb:app:attention-state-space}. Following~\eqref{ptnb:eq:routing-Gamma-pair}, we form the output difference
        $\Gamma^{O}_{t,d}$ at entry (defined in~\eqref{ptnb:eq:attn-Gamma-O})
        and the input-neuron difference at exit
        \begin{align}\label{ptnb:eq:attn-Gamma-X}
            \Gamma^{X}_{k,e} \;\coloneqq\;
            \Gamma_x\!\bigl(s_{(k,e),+}^{(\mathrm{att},\,\mathrm{X},\,\mathrm{act})}\bigr) -
            \Gamma_x\!\bigl(s_{(k,e),-}^{(\mathrm{att},\,\mathrm{X},\,\mathrm{act})}\bigr).
        \end{align}
    \item the \emph{attribution-side} object is the standard $\alpha\beta$-\ac{lrp} relevance vector
        $R^{X}_{k,e}$ given by the input-level target rule~\eqref{ptnb:eq:attn-target-input}, seeded by
        $R^{O}_{t,d}$ from the surrounding linear-layer recursion above the head.
\end{itemize}

\begin{theoremenv}[$\alpha\beta$-\ac{lrp} recovery for attention, \LeanResult{ptnb:thm:attn-ab-recovery}]\label{ptnb:thm:attn-ab-recovery}\label{thm:attention_ab_recovery}
For signed inputs $X$, under the SPNE of \S\ref{ptnb:app:attention-state-space} with reference
$\bar\pi_t = A_{t,\cdot}$ and the trajectory discount of~\eqref{ptnb:eq:routing-edge-discount}
restricted to the (sole) sign-oracle split at $s^{(\mathrm{att},\,\mathrm{O},\,\mathrm{act})}$, the
per-input-neuron player-difference $\Gamma^{X}_{k,e}$ from~\eqref{ptnb:eq:attn-Gamma-X} satisfies
\begin{align}\label{ptnb:eq:attn-ab-recovery}
    \Gamma^{X}_{k,e}
    \;=\;
    \sum_{t,d} \left(
    \alpha\, \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^+}{Z_{t,d}^+}
    \;-\;
    \beta\, \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^-}{Z_{t,d}^-}
    \right) \Gamma^{O}_{t,d},
\end{align}
where the per-output-entry input difference is
\begin{align}\label{ptnb:eq:attn-Gamma-O}
    \Gamma^{O}_{t,d} \;\coloneqq\;
    \Gamma_x\!\bigl(s_{t,d,+}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{act})}\bigr) \;-\;
    \Gamma_x\!\bigl(s_{t,d,-}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{act})}\bigr),
\end{align}
with $\Gamma^{O}_{t,d} = R^{O}_{t,d}$ by inductive hypothesis from the surrounding recursion, for
the unit seed used by the implementation. Any real seed $r_u$ multiplies the occupation difference
by $r_u$, including the signed affine-logit seed of Section~\ref{ptnb:app:affine-readout}. The right-hand side is exactly the target
rule~\eqref{ptnb:eq:attn-target-input}, hence $\Gamma^{X}_{k,e} = R^{X}_{k,e}$. Aggregating over $e$ and
the bias node at fixed $(k,d)$ collects the relevance routed through the Value entry $v_{k,d}$:
\begin{align}\label{ptnb:eq:attn-Gamma-V-corollary}
    \Gamma^{\mathrm{val}}_{k,d}
    \;=\;
    \sum_t \left( \alpha\, \frac{A_{t,k}\, \tilde{v}_{k,d}^+}{Z_{t,d}^+} \;-\; \beta\, \frac{A_{t,k}\,
    \tilde{v}_{k,d}^-}{Z_{t,d}^-} \right) \Gamma^{O}_{t,d}
    \;=\; R^{\mathrm{val}}_{k,d},
\end{align}
recovering the per-Value-entry formula~\eqref{ptnb:eq:attn-target-rule}.\end{theoremenv}

\begin{proof}
By hypothesis, the surrounding recursion supplies $\Gamma^{O}_{t,d} = R^{O}_{t,d}$ at every
output $(t,d)$; the proof is block-local.

\textbf{Per-state recursion at the input activation states.} Fix an input embedding $(k,e)$ and a
player label $r \in \{+,-\}$. A trajectory reaches
$s_{(k,e),r}^{(\mathrm{att},\,\mathrm{X},\,\mathrm{act})}$ from an output activation
$s_{t,d,p}^{(\mathrm{att},\,\mathrm{O},\,\mathrm{act})}$ in three stages, with the discounts and player switches of
\S\ref{ptnb:app:attention-state-space}. Each path fixes $d \in \{1,\dots,d_h\}$:
the oracle acts at $(t,d)$, $\pi_{t,d,\kappa}^\star$ depends on $d$ through
$\tilde{v}_{k,d}^\kappa$, and $\pi_{\mathrm{val},k,d,\kappa}^\star$ also depends on $d$.
The final transition drops $d$, reaching $X_{k,e}$ indexed by $(k,e)$;
the head-feature axis occurs downstream at the Value projection's output.
\begin{itemize}
    \item \emph{$\kappa = +$ branch (per $(t,d)$).} Sign-oracle takes $s_{t,d,p}^{(\mathrm{O},\mathrm{act})}
            \to s_{(t,d),p,+}^{(\mathrm{O},\mathrm{lin})}$ with probability $\tfrac12$ and trajectory discount
        $2\alpha$ (player retained). The Value-routing policy $\pi_{t,d,+}^\star(k)$ takes the trajectory
        to $s_{(k,d),p,+}^{(\mathrm{val},\mathrm{lin})}$ (player, sign, $d$ preserved, discount $1$). The
        Value-projection routing $\pi_{\mathrm{val},k,d,+}^\star(e)$ takes the trajectory to
        $s_{(k,e),p}^{(\mathrm{X},\mathrm{act})}$ (player preserved, $d$ dropped, discount $1$). The resulting contribution from $(t,d)$ to $\Gamma_x(s_{(k,e),r}^{(\mathrm{X},\mathrm{act})})$ is
        \begin{align}
            \tfrac12 (2\alpha)\, \pi_{t,d,+}^\star(k)\, \pi_{\mathrm{val},k,d,+}^\star(e)\,
            \Gamma_x\!\bigl(s_{t,d,r}^{(\mathrm{O},\mathrm{act})}\bigr)
            &=\; \alpha\, \frac{A_{t,k}\, \tilde{v}_{k,d}^+}{Z_{t,d}^+}\,\frac{[X_{k,e}
            W_{v,e,d}]^+}{\tilde{v}_{k,d}^+}\, \Gamma_x\!\bigl(s_{t,d,r}^{(\mathrm{O},\mathrm{act})}\bigr) \\
            &=\; \alpha\, \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^+}{Z_{t,d}^+}\,
            \Gamma_x\!\bigl(s_{t,d,r}^{(\mathrm{O},\mathrm{act})}\bigr),
        \end{align}
        where cancellation applies when $\tilde{v}_{k,d}^+>0$. If this mass or $Z_{t,d}^+$ is zero,
the ordinary-successor contribution on both sides is zero by~\eqref{ptnb:eq:attn-zero-ratio}.
    \item \emph{$\kappa = -$ branch (per $(t,d)$).} Sign-oracle takes $s_{t,d,p}^{(\mathrm{O},\mathrm{act})}
            \to s_{(t,d),p',-}^{(\mathrm{O},\mathrm{lin})}$ with probability $\tfrac12$, discount $2\beta$,
        turn switched to opponent $p'$; the rest of the path preserves $p'$, ending at
        $s_{(k,e),p'}^{(\mathrm{X},\mathrm{act})}$. The contribution to
        $\Gamma_x(s_{(k,e),r}^{(\mathrm{X},\mathrm{act})})$ comes from sources with player $p = r'$ (so the
        switch yields $r$):
        \begin{align}
            \tfrac12 (2\beta)\, \pi_{t,d,-}^\star(k)\, \pi_{\mathrm{val},k,d,-}^\star(e)\,
            \Gamma_x\!\bigl(s_{t,d,r'}^{(\mathrm{O},\mathrm{act})}\bigr)
            \;=\; \beta\, \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^-}{Z_{t,d}^-}\,
            \Gamma_x\!\bigl(s_{t,d,r'}^{(\mathrm{O},\mathrm{act})}\bigr).
        \end{align}
\end{itemize}
Summing over $(t,d)$ and over the two branches,
\begin{align}\label{ptnb:eq:attn-Gamma-X-recursion}
    \Gamma_x\!\bigl(s_{(k,e),r}^{(\mathrm{att},\,\mathrm{X},\,\mathrm{act})}\bigr)
    \;=\; \sum_{t,d} \Bigl[
    \alpha\, \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^+}{Z_{t,d}^+}\,
    \Gamma_x\!\bigl(s_{t,d,r}^{(\mathrm{O},\mathrm{act})}\bigr)
    \;+\;
    \beta\, \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^-}{Z_{t,d}^-}\,
    \Gamma_x\!\bigl(s_{t,d,r'}^{(\mathrm{O},\mathrm{act})}\bigr)
    \Bigr],
\end{align}
where the $\tfrac12$ from the sign-oracle splits cancels against the $2$ in $2\alpha, 2\beta$.

\textbf{Player difference.} Differencing~\eqref{ptnb:eq:attn-Gamma-X-recursion} for $r{=}+$ minus
$r{=}-$ uses $\Gamma_x(s_{t,d,+}^{(\mathrm{O},\mathrm{act})}) -
    \Gamma_x(s_{t,d,-}^{(\mathrm{O},\mathrm{act})}) = \Gamma^{O}_{t,d}$ in the $+$ branch (player kept)
and the opposite $-\Gamma^{O}_{t,d}$ in the $-$ branch (player switched):
\begin{align}
    \Gamma^{X}_{k,e}
    \;=\; \alpha\, \sum_{t,d} \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^+}{Z_{t,d}^+}\, \Gamma^{O}_{t,d}
    \;-\; \beta\, \sum_{t,d} \frac{A_{t,k}\, [X_{k,e} W_{v,e,d}]^-}{Z_{t,d}^-}\, \Gamma^{O}_{t,d},
\end{align}
which is~\eqref{ptnb:eq:attn-ab-recovery}.

\textbf{Target-rule identification.} With $\Gamma^{O}_{t,d} = R^{O}_{t,d}$, the preceding identity
matches~\eqref{ptnb:eq:attn-target-input}, giving $\Gamma^{X}_{k,e} = R^{X}_{k,e}$. Aggregating
over $e$ and the bias node at fixed $(k,d)$ uses $\sum_e [X_{k,e} W_{v,e,d}]^\kappa +
    [b_{v,d}]^\kappa = \tilde{v}_{k,d}^\kappa$, giving~\eqref{ptnb:eq:attn-Gamma-V-corollary}, which
matches~\eqref{ptnb:eq:attn-target-rule}.
\end{proof}

\section{Risk-Averse Attention via ADF}\label{ptnb:app:risk-attn}

When \ac{ptnbADF} variance propagation is active, let $s_{tk}$ denote the uncertainty score attached to
attending from query token $t$ to key token $k$ (in the current implementation, $s_{tk} =
    \hat{\sigma}_k$, the square root of the feature-averaged \ac{ptnbADF} variance of token $k$'s query
projection; it depends only on the key token and is a proxy for, not the standard deviation of,
$e_{tk}$). Risk-averse attention shifts the QK-logits in the direction of reduced uncertainty,
\begin{align}\label{ptnb:eq:risk-attn-policy}
    \tilde e_{tk} = e_{tk} + \lambda_{\mathrm{sm}}\, s_{tk},
    \qquad \lambda_{\mathrm{sm}} \le 0,
\end{align}
yielding the risk-modified policy
\begin{align}\label{ptnb:eq:risk-attn-policy-softmax}
    \tilde{\pi}_t(k) = \frac{\exp(\tilde e_{tk})}{\sum_{k'} \exp(\tilde e_{tk'})}.
\end{align}
The sign convention $\lambda_{\mathrm{sm}} \le 0$ is used throughout the paper and in all tables:
more-negative values penalize high-uncertainty keys more strongly, so the shift always redirects
relevance toward tokens whose QK-interactions are stable under input noise. The shift is computed
in the AV-backward hook from cached pre-softmax logits, so the forward pass and model predictions
remain unchanged.

The naming \emph{softmax-shift risk aversion} is a deliberate analog of the activation-side risk
aversion $\lambda$ but acts in a different direction: $\lambda$ is risk-averse on the value side
(it routes through stable activations by lowering mean$-|\lambda|\sigma$), whereas
$\lambda_{\mathrm{sm}}<0$ exponentially tilts the routing policy, $\tilde\pi_t(k) \propto
    A_{t,k}\exp(\lambda_{\mathrm{sm}} s_{tk})$, toward \emph{low}-uncertainty keys: $\sum_k
    \tilde\pi_t(k)\, s_{tk}$ is non-decreasing in $\lambda_{\mathrm{sm}}$, while the entropy of
$\tilde\pi_t$ may rise or fall. We retain the ``risk-averse'' label because what is being avoided
is uncertainty in the routed pathway, but the reader should note that the shift acts on the routing
policy, not on the player's utility curvature.
