Thesis.TrajectoryHellinger.trajectory_hellinger_recursion

Locally checked, unpublished Lean source. This is an addition to the pinned thesis formalization; it has not been committed or published as a GitHub revision.

Hellinger distance of normalized prefix and complete trajectory laws equals the square root of one minus the total recursively propagated geometric-mean overlap. Layer state types may vary; zero probabilities are allowed. Lean counts elapsed steps n=L-l; n=L gives the full trajectory law.

3/-!
4Hellinger distance of actual layered trajectory laws, computed by transporting
5geometric-mean overlap mass. The state type may vary at every layer. We index
6the traversal by elapsed steps `n`; in the thesis notation this is `n = L - l`,
7with the full trajectory law obtained at `n = L`. A cemetery can be included
8in each state type, but absorption is not required by the distance identity.
9-/

16/-- A prefix includes its initial state and every subsequently visited state. -/
17def Prefix (S : ℕ → Type) : ℕ → Type
18  | 0 => S 0
19  | n + 1 => Prefix S n × S (n + 1)
20
21instance prefixFintype (S : ℕ → Type) [∀ n, Fintype (S n)] :
22    (n : ℕ) → Fintype (Prefix S n)
23  | 0 => inferInstanceAs (Fintype (S 0))
24  | n + 1 => @instFintypeProd (Prefix S n) (S (n + 1))
25      (prefixFintype S n) inferInstance
26
27def last {S : ℕ → Type} : (n : ℕ) → Prefix S n → S n
28  | 0, p => p
29  | _ + 1, p => p.2
30
31abbrev Kernel (S : ℕ → Type) := (n : ℕ) → S n → S (n + 1) → ℝ
32
33variable {S : ℕ → Type} [∀ n, Fintype (S n)]
34
35/-- Independent product law of all prefixes, including the common delta start. -/
36def pathLaw (K : Kernel S) (s₀ : S 0) : (n : ℕ) → Prefix S n → ℝ
37  | 0, p => if p = s₀ then 1 else 0
38  | n + 1, p => pathLaw K s₀ n p.1 * K n (last n p.1) p.2
39
40/-- Forward transport, independently defined without enumerating paths. -/
41def mass (K : Kernel S) (s₀ : S 0) : (n : ℕ) → S n → ℝ
42  | 0, s => if s = s₀ then 1 else 0
43  | n + 1, t => ∑ s, mass K s₀ n s * K n s t
44
45def geometricKernel (A B : Kernel S) : Kernel S :=
46  fun n s t => Real.sqrt (A n s t * B n s t)
47
48/-- The thesis quantity g, initialized at the start and propagated edgewise. -/
49def g (A B : Kernel S) (s₀ : S 0) (n : ℕ) : S n → ℝ :=
50  mass (geometricKernel A B) s₀ n
51
52/-- Computed overlap mass; identification with path overlap is proved below. -/
53def BC (A B : Kernel S) (s₀ : S 0) (n : ℕ) : ℝ := ∑ s, g A B s₀ n s
54
55/-- Squared Hellinger distance is defined from differences of square roots. -/
56def hellingerSquared {X : Type} [Fintype X] (p q : X → ℝ) : ℝ :=
57  (1 / 2 : ℝ) * ∑ x, (Real.sqrt (p x) - Real.sqrt (q x)) ^ 2
58
59/-- The distance, independently of any overlap recursion. -/
60def hellinger {X : Type} [Fintype X] (p q : X → ℝ) : ℝ :=
61  Real.sqrt (hellingerSquared p q)
62
63theorem hellingerSquared_nonneg {X : Type} [Fintype X] (p q : X → ℝ) :
64    0 ≤ hellingerSquared p q := by
65  unfold hellingerSquared
66  exact mul_nonneg (by norm_num) (Finset.sum_nonneg fun x _ => sq_nonneg _)
67
68theorem hellinger_sq {X : Type} [Fintype X] (p q : X → ℝ) :
69    hellinger p q ^ 2 = hellingerSquared p q := by
70  exact Real.sq_sqrt (hellingerSquared_nonneg p q)

73theorem pathLaw_nonnegative (K : Kernel S) (s₀ : S 0)
74    (n : ℕ) (hK : ∀ i < n, ∀ s t, 0 ≤ K i s t) (p : Prefix S n) :
75    0 ≤ pathLaw K s₀ n p := by
76  classical
77  induction n with
78  | zero =>
79      change 0 ≤ if p = s₀ then (1 : ℝ) else 0
80      split_ifs <;> norm_num
81  | succ n ih =>
82      exact mul_nonneg (ih (fun i hi => hK i (Nat.lt_succ_of_lt hi)) p.1)
83        (hK n (Nat.lt_succ_self n) (last n p.1) p.2)
84
85theorem pathLaw_normalized (K : Kernel S) (s₀ : S 0)
86    (n : ℕ) (hKrow : ∀ i < n, ∀ s, ∑ t, K i s t = 1) :
87    (∑ p : Prefix S n, pathLaw K s₀ n p) = 1 := by
88  classical
89  induction n with
90  | zero =>
91      change (∑ p : S 0, if p = s₀ then (1 : ℝ) else 0) = 1
92      simp
93  | succ n ih =>
94    change (∑ p : Prefix S n × S (n + 1),
95      pathLaw K s₀ n p.1 * K n (last n p.1) p.2) = 1
96    rw [Fintype.sum_prod_type]
97    simp only [← Finset.mul_sum, hKrow n (Nat.lt_succ_self n), mul_one]
98    exact ih (fun i hi => hKrow i (Nat.lt_succ_of_lt hi))

100/-- The transported mass equals the sum over paths with the given endpoint. -/
101theorem mass_eq_endpoint_sum (K : Kernel S) (s₀ : S 0) (n : ℕ) (t : S n) :
102    mass K s₀ n t = ∑ p : Prefix S n,
103      if last n p = t then pathLaw K s₀ n p else 0 := by
104  classical
105  induction n with
106  | zero =>
107      change (if t = s₀ then (1 : ℝ) else 0) =
108        ∑ p : S 0, if p = t then (if p = s₀ then 1 else 0) else 0
109      simp
110  | succ n ih =>
111    change (∑ s, mass K s₀ n s * K n s t) =
112      ∑ p : Prefix S n × S (n + 1),
113        if p.2 = t then pathLaw K s₀ n p.1 * K n (last n p.1) p.2 else 0
114    rw [Fintype.sum_prod_type]
115    simp only [Finset.sum_ite_eq', Finset.mem_univ, ite_true]
116    simp_rw [ih, Finset.sum_mul]
117    rw [Finset.sum_comm]
118    apply Finset.sum_congr rfl
119    intro p _
120    simp only [ite_mul, zero_mul]
121    simp
122
123theorem mass_total_eq_path_sum (K : Kernel S) (s₀ : S 0) (n : ℕ) :
124    (∑ s, mass K s₀ n s) = ∑ p : Prefix S n, pathLaw K s₀ n p := by
125  simp_rw [mass_eq_endpoint_sum]
126  rw [Finset.sum_comm]
127  simp

130/-- Geometric means factor along the actual varying-state prefix law. -/
131theorem pathLaw_geometric (A B : Kernel S) (s₀ : S 0)
132    (n : ℕ) (hA : ∀ i < n, ∀ s t, 0 ≤ A i s t)
133    (hB : ∀ i < n, ∀ s t, 0 ≤ B i s t) (p : Prefix S n) :
134    Real.sqrt (pathLaw A s₀ n p * pathLaw B s₀ n p) =
135      pathLaw (geometricKernel A B) s₀ n p := by
136  classical
137  induction n with
138  | zero =>
139      change Real.sqrt ((if p = s₀ then (1 : ℝ) else 0) *
140        (if p = s₀ then (1 : ℝ) else 0)) = if p = s₀ then 1 else 0
141      split_ifs <;> norm_num
142  | succ n ih =>
143    simp only [pathLaw]
144    have he : (pathLaw A s₀ n p.1 * A n (last n p.1) p.2) *
145        (pathLaw B s₀ n p.1 * B n (last n p.1) p.2) =
146        (pathLaw A s₀ n p.1 * pathLaw B s₀ n p.1) *
147        (A n (last n p.1) p.2 * B n (last n p.1) p.2) := by ring
148    have hAn := fun i hi => hA i (Nat.lt_succ_of_lt hi)
149    have hBn := fun i hi => hB i (Nat.lt_succ_of_lt hi)
150    rw [he, Real.sqrt_mul (mul_nonneg
151      (pathLaw_nonnegative A s₀ n hAn p.1) (pathLaw_nonnegative B s₀ n hBn p.1)),
152      ih hAn hBn]
153    rfl

155/-- The computed total mass is the overlap of whole prefixes, not merely endpoints. -/
156theorem BC_eq_path_overlap (A B : Kernel S) (s₀ : S 0)
157    (n : ℕ) (hA : ∀ i < n, ∀ s t, 0 ≤ A i s t)
158    (hB : ∀ i < n, ∀ s t, 0 ≤ B i s t) :
159    BC A B s₀ n = Thesis.Games.overlap (pathLaw A s₀ n) (pathLaw B s₀ n) := by
160  unfold BC g Thesis.Games.overlap
161  rw [mass_total_eq_path_sum]
162  simp_rw [pathLaw_geometric A B s₀ n hA hB]

164/-- Remove the last k coordinates of a longer path. -/
165def truncate (n : ℕ) : (k : ℕ) → Prefix S (n + k) → Prefix S n
166  | 0, p => p
167  | k + 1, p => truncate n k p.1
168
169/-- Normalized future transitions make the shorter law the true marginal
170of the longer trajectory distribution. -/
171theorem prefix_marginal (K : Kernel S) (s₀ : S 0)
172    (n k : ℕ) (hKrow : ∀ i < n + k, ∀ s, ∑ t, K i s t = 1)
173    (q : Prefix S n) :
174    (∑ p : Prefix S (n + k),
175      if truncate n k p = q then pathLaw K s₀ (n + k) p else 0) =
176      pathLaw K s₀ n q := by
177  classical
178  induction k with
179  | zero =>
180      change (∑ p : Prefix S n, if p = q then pathLaw K s₀ n p else 0) = _
181      exact Fintype.sum_ite_eq' q (pathLaw K s₀ n)
182  | succ k ih =>
183    change (∑ p : Prefix S (n + k) × S (n + k + 1),
184      if truncate n k p.1 = q then
185        pathLaw K s₀ (n + k) p.1 * K (n + k) (last (n + k) p.1) p.2
186      else 0) = _
187    rw [Fintype.sum_prod_type]
188    have hs (p : Prefix S (n + k)) :
189        (∑ t, if truncate n k p = q then
190          pathLaw K s₀ (n + k) p * K (n + k) (last (n + k) p) t else 0) =
191        if truncate n k p = q then pathLaw K s₀ (n + k) p else 0 := by
192      by_cases hp : truncate n k p = q
193      · simp [hp, ← Finset.mul_sum, hKrow (n + k) (Nat.lt_succ_self (n + k))]
194      · simp [hp]
195    simp_rw [hs]
196    exact ih (fun i hi => hKrow i (Nat.lt_succ_of_lt hi))

198/-- The exact displayed Hellinger identities for every prefix length n,
199including the full law at n=L and the common-start boundary n=0.
200The equation uses layer index l=L-n; BC and g are the independently computed
201overlap recursion, and H is defined from square-root differences of path laws. -/
202theorem trajectory_hellinger_recursion (A B : Kernel S) (s₀ : S 0)
203    (n : ℕ) (hA : ∀ i < n, ∀ s t, 0 ≤ A i s t)
204    (hB : ∀ i < n, ∀ s t, 0 ≤ B i s t)
205    (hArow : ∀ i < n, ∀ s, ∑ t, A i s t = 1)
206    (hBrow : ∀ i < n, ∀ s, ∑ t, B i s t = 1) :
207    hellingerSquared (pathLaw A s₀ n) (pathLaw B s₀ n) =
208        hellinger (pathLaw A s₀ n) (pathLaw B s₀ n) ^ 2 ∧
209    hellinger (pathLaw A s₀ n) (pathLaw B s₀ n) ^ 2 = 1 - BC A B s₀ n ∧
210    hellinger (pathLaw A s₀ n) (pathLaw B s₀ n) = Real.sqrt (1 - BC A B s₀ n) ∧
211    hellinger (pathLaw A s₀ n) (pathLaw B s₀ n) =
212        Real.sqrt (1 - ∑ s, g A B s₀ n s) := by
213  have hs : hellingerSquared (pathLaw A s₀ n) (pathLaw B s₀ n) =
214      1 - BC A B s₀ n := by
215    rw [BC_eq_path_overlap A B s₀ n hA hB]
216    exact Thesis.Games.hellinger_identity _ _
217      (pathLaw_nonnegative A s₀ n hA) (pathLaw_nonnegative B s₀ n hB)
218      (pathLaw_normalized A s₀ n hArow) (pathLaw_normalized B s₀ n hBrow)
219  have hd : hellinger (pathLaw A s₀ n) (pathLaw B s₀ n) =
220      Real.sqrt (1 - BC A B s₀ n) := by
221    unfold hellinger
222    rw [hs]
223  exact ⟨(hellinger_sq _ _).symm, (hellinger_sq _ _).trans hs, hd, hd⟩

225theorem common_start_distance (A B : Kernel S) (s₀ : S 0) :
226    hellinger (pathLaw A s₀ 0) (pathLaw B s₀ 0) = 0 := by
227  simp [hellinger, hellingerSquared, Prefix, pathLaw]
